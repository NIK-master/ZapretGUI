using System;
using System.IO;
using System.Threading.Tasks;
using System.Windows;
using Microsoft.Web.WebView2.Core;
using ZapretGUI.Core;

namespace ZapretGUI
{
    public partial class MainWindow : Window
    {
        private readonly TrayIconManager _trayIconManager;
        private WebBridge? _webBridge;

        public MainWindow()
        {
            InitializeComponent();

            var configPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "config.json");
            var isFirstRun = !File.Exists(configPath);

            SettingsManager.Load();
            UIHelper.ApplyThemeColors();

            if (isFirstRun)
            {
                var wizard = new Views.WizardWindow();
                wizard.ShowDialog();
                SettingsManager.Save();
            }

            _trayIconManager = new TrayIconManager(this);

            _ = InitializeAppAsync();
        }

        private async Task InitializeAppAsync()
        {
            var modManager = new ModManager();
            modManager.InitializeFolders();
            await modManager.SyncActiveBatModsAsync();
            await modManager.ApplyListModsAsync();

            await InitializeWebViewAsync();

            BypassController.Current.NetMonitor.Start();
            _ = CheckUpdatesOnStartupAsync();
        }

        private async Task InitializeWebViewAsync()
        {
            var userDataFolder = Path.Combine(Path.GetTempPath(), "ZapretGUI_WebView2");
            var env = await CoreWebView2Environment.CreateAsync(null, userDataFolder);
            await MainWebView.EnsureCoreWebView2Async(env);

            MainWebView.CoreWebView2.Settings.AreDefaultContextMenusEnabled = false;
            MainWebView.CoreWebView2.Settings.IsZoomControlEnabled = false;

            _webBridge = new WebBridge(MainWebView, this);
            MainWebView.CoreWebView2.WebMessageReceived += (s, e) =>
            {
                _webBridge.HandleMessage(e.WebMessageAsJson);
            };

            // 1. Ищем index.html рядом с .exe (для Release)
            var baseDir = AppDomain.CurrentDomain.BaseDirectory;
            var htmlPath = Path.Combine(baseDir, "wwwroot", "index.html");

            // 2. Если в bin еще не скопировалось — ищем в корне проекта (удобно для Debug)
            if (!File.Exists(htmlPath))
            {
                var projectRootHtml = Path.GetFullPath(Path.Combine(baseDir, @"..\..\..\wwwroot\index.html"));
                if (File.Exists(projectRootHtml))
                    htmlPath = projectRootHtml;
            }

            // 3. Также проверяем случай, если index.html лежит прямо в корне проекта
            if (!File.Exists(htmlPath))
            {
                var rootFallback = Path.GetFullPath(Path.Combine(baseDir, @"..\..\..\index.html"));
                if (File.Exists(rootFallback))
                    htmlPath = rootFallback;
            }

            if (File.Exists(htmlPath))
            {
                MainWebView.CoreWebView2.Navigate(new Uri(htmlPath).AbsoluteUri);
            }
            else
            {
                MainWebView.NavigateToString($@"
                    <body style='background:#09090B;color:white;font-family:sans-serif;padding:40px;'>
                        <h2 style='color:#EF4444;'>Файл wwwroot/index.html не найден!</h2>
                        <p>Убедитесь, что в проекте создана папка <b>wwwroot</b> и внутри неё лежит файл <b>index.html</b>.</p>
                        <p style='color:#888;font-family:monospace;'>Ожидаемый путь: {htmlPath}</p>
                    </body>");
            }
        }

        private async Task CheckUpdatesOnStartupAsync()
        {
            var stopServicesAction = () => { if (IsBypassRunning()) ToggleBypass(); };
            var progress = new Progress<string>(status =>
                _webBridge?.SendEvent("log", new { message = $"[Updater] {status}", time = DateTime.Now.ToString("HH:mm:ss") }));
            await UpdateManager.RunUpdateWorkflowAsync(isManualCheck: false, stopServicesAction, progress);
        }

        // Методы совместимости для системного трея (TrayIconManager)
        public void UpdateIndicators(bool isZapretRunning, bool isProxyRunning)
        {
            _webBridge?.SendEvent("bypassState", new
            {
                isRunning = isZapretRunning || isProxyRunning,
                zapretRunning = isZapretRunning,
                proxyRunning = isProxyRunning
            });
        }

        public void UpdateNetworkIndicator(bool isOnline)
        {
            _webBridge?.SendEvent("netStatus", new { isOnline });
        }

        public void ShowNotification(string title, string message, System.Windows.Forms.ToolTipIcon icon = System.Windows.Forms.ToolTipIcon.Info)
        {
            _trayIconManager.ShowNotification(title, message, icon);
        }

        private void Window_Closing(object sender, System.ComponentModel.CancelEventArgs e)
        {
            if (SettingsManager.Current.MinimizeOnClose)
            {
                e.Cancel = true;
                this.Hide();
                ShowNotification("Программа работает в фоне", "Zapret свернут в системный трей. Дважды кликните по иконке в трее, чтобы открыть окно.");
            }
        }

        public bool IsBypassRunning() => BypassController.Current.IsRunning;

        public async void ToggleBypass()
        {
            if (BypassController.Current.IsRunning)
                BypassController.Current.StopServices();
            else
                await BypassController.Current.StartServicesAsync(
                    SettingsManager.Current.SelectedProfileName,
                    SettingsManager.Current.ZapretEnabled,
                    SettingsManager.Current.TgProxyEnabled);

            UpdateIndicators(BypassController.Current.Zapret.IsRunning, BypassController.Current.TgProxy.IsRunning);
        }

        public async Task RestartBypassAsync()
        {
            var profile = SettingsManager.Current.SelectedProfileName;
            await BypassController.Current.RestartServicesAsync(profile);
            UpdateIndicators(BypassController.Current.Zapret.IsRunning, BypassController.Current.TgProxy.IsRunning);
        }

        public void AnimateWindowSize(bool isCompact)
        {
            Dispatcher.Invoke(() =>
            {
                this.Width = isCompact ? 850 : 1100;
                this.Height = isCompact ? 520 : 760;
            });
        }

        protected override void OnClosed(EventArgs e)
        {
            BypassController.Current.NetMonitor.Stop();
            _trayIconManager.Dispose();
            base.OnClosed(e);
        }
    }
}

using System;
using System.Collections.Generic;
using System.Threading.Tasks;

namespace ZapretGUI.Core
{
    public class BypassController
    {
        public static BypassController Current { get; } = new BypassController();

        public ZapretManager Zapret { get; } = new ZapretManager();
        public TgProxyManager TgProxy { get; } = new TgProxyManager();
        public NetworkMonitor NetMonitor { get; } = new NetworkMonitor();
        public ZapretScanner Scanner { get; } = new ZapretScanner();

        private readonly List<IBypassService> _services;

        public bool IsRunning => Zapret.IsRunning || TgProxy.IsRunning;

        public event Action<string>? OnLog;

        private BypassController()
        {
            _services = new List<IBypassService> { Zapret, TgProxy };

            foreach (var service in _services)
            {
                service.LogMessage += msg => OnLog?.Invoke(msg);
            }
            Scanner.LogMessage += msg => OnLog?.Invoke(msg);
        }

        public async Task StartServicesAsync(string profileName, bool useZapret, bool useTgProxy)
        {
            if (useZapret)
            {
                OnLog?.Invoke($"[Zapret] Подготовка профиля {profileName}...");
                await Task.Delay(400);
                OnLog?.Invoke($"[Zapret] Запуск службы...");
                Zapret.Start(profileName);
            }

            if (useTgProxy)
            {
                OnLog?.Invoke("[TgWsProxy] Настройка маршрутов...");
                await Task.Delay(300);
                OnLog?.Invoke("[TgWsProxy] Запуск прокси...");
                TgProxy.Start();
            }
        }

        public void StopServices()
        {
            foreach (var service in _services)
            {
                service.Stop();
            }
            OnLog?.Invoke("🛑 Все модули остановлены.");
        }

        public async Task RestartServicesAsync(string profileName)
        {
            StopServices();
            await Task.Delay(1000);

            if (SettingsManager.Current.ZapretEnabled)
                Zapret.Start(profileName);

            if (SettingsManager.Current.TgProxyEnabled)
                TgProxy.Start();

            OnLog?.Invoke("✅ Службы успешно перезапущены.");
        }

        public async Task<long> CheckPingAsync()
        {
            var hostToPing = AppConstants.AwsPingHost;
            var portToPing = AppConstants.AwsPingPort;

            var fullUrl = SettingsManager.Current.PingUrl;
            if (!string.IsNullOrWhiteSpace(fullUrl))
            {
                if (Uri.TryCreate(fullUrl, UriKind.Absolute, out Uri? uri))
                {
                    hostToPing = uri.Host;
                    portToPing = uri.Port > 0 ? uri.Port : 443;
                }
                else
                    hostToPing = fullUrl;
            }

            return await NetworkHelper.TcpPingAsync(hostToPing, portToPing);
        }
    }
}

        public static async Task<(bool ok, double latencyMs, string error)> TcpConnectAsync(string ip, int port, double timeoutSec = 3.0)
        {
            var sw = Stopwatch.StartNew();
            try
            {
                using var tcp = new TcpClient();
                using var cts = new CancellationTokenSource(TimeSpan.FromSeconds(timeoutSec));
                await tcp.ConnectAsync(ip, port, cts.Token);
                return (true, sw.Elapsed.TotalMilliseconds, "");
            }
            catch (OperationCanceledException) { return (false, 0, "timeout"); }
            catch (SocketException e) when (e.SocketErrorCode == SocketError.ConnectionRefused) { return (false, 0, "refused"); }
            catch (Exception e) { return (false, 0, e.Message); }
        }

        private static async Task<string> TlsHandshakeAsync(string ip, string sni, double timeoutSec = 4.0)
        {
            try
            {
                using var tcp = new TcpClient();
                using var cts = new CancellationTokenSource(TimeSpan.FromSeconds(timeoutSec));
                await tcp.ConnectAsync(ip, 443, cts.Token);

                using var ssl = new SslStream(tcp.GetStream(), false, (_, _, _, _) => true);
                await ssl.AuthenticateAsClientAsync(new SslClientAuthenticationOptions
                {
                    TargetHost = sni,
                    RemoteCertificateValidationCallback = (_, _, _, _) => true,
                }, cts.Token);
                return "ok";
            }
            catch (OperationCanceledException) { return "timeout"; }
            catch (Exception e)
            {
                var msg = e.Message.ToLower();
                if (msg.Contains("connection reset") || msg.Contains("forcibly closed") || msg.Contains("eof"))
                    return "rst";
                return $"error:{e.Message}";
            }
        }

        private static async Task<List<string>> DnsResolveAsync(string domain, string? server = null)
        {
            if (server is null)
            {
                try
                {
                    var entry = await Dns.GetHostEntryAsync(domain);
                    return entry.AddressList.Select(a => a.ToString()).Distinct().ToList();
                }
                catch { return []; }
            }

            try
            {
                var txId = (ushort)Random.Shared.Next(0, 65536);
                var query = BuildDnsQuery(domain, txId);
                using var udp = new UdpClient();
                udp.Client.ReceiveTimeout = 3000;
                var ep = new IPEndPoint(IPAddress.Parse(server), 53);
                await udp.SendAsync(query, ep);
                var result = await udp.ReceiveAsync();
                return ParseDnsResponse(result.Buffer);
            }
            catch { return []; }
        }

        private static byte[] BuildDnsQuery(string domain, ushort txId)
        {
            var buf = new List<byte>
            {
                (byte)(txId >> 8), (byte)(txId & 0xff),
                0x01, 0x00, 0x00, 0x01, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00
            };
            foreach (var part in domain.Split('.'))
            {
                buf.Add((byte)part.Length);
                buf.AddRange(Encoding.ASCII.GetBytes(part));
            }
            buf.Add(0);
            buf.AddRange(new byte[] { 0x00, 0x01, 0x00, 0x01 });
            return buf.ToArray();
        }

        private static List<string> ParseDnsResponse(byte[] data)
        {
            var ips = new List<string>();
            try
            {
                var ancount = (data[6] << 8) | data[7];
                if (ancount == 0) 
                    return ips;

                var i = 12;
                while (i < data.Length && data[i] != 0) 
                    i += data[i] + 1;

                i += 5;
                for (var a = 0; a < ancount && i + 10 < data.Length; a++)
                {
                    i += 2;
                    var rtype = (data[i] << 8) | data[i + 1];
                    var rdlen = (data[i + 8] << 8) | data[i + 9];
                    i += 10;
                    if (rtype == 1 && rdlen == 4 && i + 4 <= data.Length)
                        ips.Add($"{data[i]}.{data[i + 1]}.{data[i + 2]}.{data[i + 3]}");
                    i += rdlen;
                }
            }
            catch { }
            return ips;
        }

        private static async Task<(bool ok, double latencyMs)> PingOnceAsync(string host)
        {
            try
            {
                using var ping = new Ping();
                var reply = await ping.SendPingAsync(host, 2000);
                if (reply.Status == IPStatus.Success)
                    return (true, reply.RoundtripTime);
            }
            catch { }
            return (false, 0);
        }

        private static async Task<bool> UdpProbeAsync(string host, int port, int timeoutMs = 2000)
        {
            try
            {
                using var udp = new UdpClient();
                await udp.SendAsync(new byte[4], new IPEndPoint(IPAddress.Parse(host), port));
                using var cts = new CancellationTokenSource(timeoutMs);
                try
                {
                    await udp.ReceiveAsync(cts.Token);
                    return true;
                }
                catch (OperationCanceledException)
                {
                    return false;
                }
                catch (SocketException e) when (e.SocketErrorCode == SocketError.ConnectionReset)
                {
                    return true;
                }
                catch { return false; }
            }
            catch { return false; }
        }

        public static AppStatus CheckAppStatus()
        {
            var status = new AppStatus();
            try
            {
                var names = Process.GetProcesses()
                    .Select(p => { try { return p.ProcessName.ToLower(); } catch { return ""; } })
                    .Where(n => n.Length > 0).ToList();

                string Match(string key) =>
                    names.FirstOrDefault(n => AppSignatures[key].Any(s => n.Contains(s))) ?? "";

                var tg = Match("telegram"); if (tg != "") { status.TelegramRunning = true; status.TelegramProcName = tg; }
                var dc = Match("discord"); if (dc != "") { status.DiscordRunning = true; status.DiscordProcName = dc; }
                var zp = Match("zapret"); if (zp != "") { status.ZapretRunning = true; status.ZapretProcName = zp; }
                var tw = Match("tgwsproxy"); if (tw != "") { status.TgWsProxyRunning = true; status.TgWsProxyProcName = tw; }
            }
            catch { }
            return status;
        }

        public static async Task<List<DcResult>> CheckTelegramDcsAsync(Action<double>? progress = null)
        {
            var done = 0;
            var total = TelegramDcs.Length * TelegramPorts.Length;
            var tasks = (from dc in TelegramDcs
                         from port in TelegramPorts
                         select Task.Run(async () =>
                         {
                             var (ok, lat, err) = await TcpConnectAsync(dc.ip, port, 4.0);
                             Interlocked.Increment(ref done);
                             progress?.Invoke((double)done / total);
                             return new DcResult
                             {
                                 DcId = dc.dcId,
                                 Ip = dc.ip,
                                 Port = port,
                                 Ok = ok,
                                 LatencyMs = ok ? lat : null,
                                 Error = err
                             };
                         })).ToList();

            var results = await Task.WhenAll(tasks);
            return [.. results.OrderBy(r => r.DcId).ThenBy(r => r.Port)];
        }

        public static async Task<List<PingResult>> CheckPingAsync(Action<double>? progress = null)
        {
            var results = new List<PingResult>();
            for (var i = 0; i < PingHosts.Length; i++)
            {
                var (ok, rtt) = await PingOnceAsync(PingHosts[i]);
                results.Add(new PingResult
                {
                    Host = PingHosts[i],
                    Ok = ok,
                    LatencyMs = ok ? rtt : null,
                    PacketLoss = ok ? 0 : 100
                });
                progress?.Invoke((double)(i + 1) / PingHosts.Length);
            }
            return results;
        }

        public static async Task<DnsResult> CheckDnsAsync(string domain = "telegram.org", Action<double>? progress = null)
        {
            var result = new DnsResult { Domain = domain };
            progress?.Invoke(0.1);
            result.SystemIps = await DnsResolveAsync(domain);
            progress?.Invoke(0.5);
            result.DohIps = await DnsResolveAsync(domain, "1.1.1.1");
            progress?.Invoke(0.9);
            if (result.SystemIps.Count > 0 && result.DohIps.Count > 0)
            {
                if (!result.SystemIps.Intersect(result.DohIps).Any())
                    result.Spoofed = true;
            }
            else if (result.SystemIps.Count == 0) 
                result.Spoofed = true;
            else if (result.DohIps.Count == 0) 
                result.Error = "1.1.1.1 недоступен";
            progress?.Invoke(1.0);
            return result;
        }

        public static async Task<DpiResult> CheckDpiAsync(Action<double>? progress = null)
        {
            var result = new DpiResult();
            const string targetIp = "149.154.167.51";
            progress?.Invoke(0.1);
            result.SniTelegram = await TlsHandshakeAsync(targetIp, "telegram.org", 5.0);
            progress?.Invoke(0.5);
            result.SniNeutral = await TlsHandshakeAsync(targetIp, "example.com", 5.0);
            progress?.Invoke(0.9);
            if (result.SniTelegram is "rst" or "timeout" && result.SniNeutral != "rst")
                result.DpiDetected = true;
            progress?.Invoke(1.0);
            return result;
        }

        public static async Task<MediaThrottleResult> CheckMediaThrottleAsync(Action<double>? progress = null)
        {
            var result = new MediaThrottleResult();
            string[] urls =
            [
                "https://cdn1.telegram.org/file/811140210/1/iQITMsUeOrg/0c6eeff1e0b33e29c0",
                "https://cdn4.telegram.org/file/811140210/1/iQITMsUeOrg/0c6eeff1e0b33e29c0",
                "https://web.telegram.org/a/runtime.js",
            ];
            progress?.Invoke(0.1);

            foreach (var url in urls)
            {
                try
                {
                    var sw = Stopwatch.StartNew();
                    var data = await _httpClient.GetByteArrayAsync(url);
                    sw.Stop();
                    var sizeKb = data.Length / 1024.0;
                    if (sizeKb < 5) continue;
                    var elapsed = Math.Max(sw.Elapsed.TotalSeconds, 0.001);
                    result.SpeedKbps = sizeKb / elapsed * 8;
                    result.Throttled = result.SpeedKbps < MediaThrottleResult.SlowThreshold;
                    progress?.Invoke(1.0);
                    return result;
                }
                catch (Exception e) { result.Error = e.Message; }
            }
            progress?.Invoke(1.0);
            return result;
        }

        public static async Task<UdpResult> CheckDiscordUdpAsync(Action<double>? progress = null)
        {
            var result = new UdpResult();
            var tested = DiscordUdpPorts.Take(5).ToList();
            result.TestedPorts = tested;
            var done = 0;

            var tasks = tested.Select(async port =>
            {
                var open = await UdpProbeAsync(DiscordHost, port);
                Interlocked.Increment(ref done);
                progress?.Invoke((double)done / tested.Count);
                return (port, open);
            });

            var results = await Task.WhenAll(tasks);
            result.OpenPorts = results.Where(r => r.open).Select(r => r.port).ToList();
            result.Blocked = result.OpenPorts.Count == 0;
            return result;
        }

        public static async Task<List<DiscordPingResult>> CheckDiscordPingAsync(Action<double>? progress = null)
        {
            var results = new List<DiscordPingResult>();
            for (var i = 0; i < DiscordPingHosts.Length; i++)
            {
                var (label, ip) = DiscordPingHosts[i];
                var (ok, lat, err) = await TcpConnectAsync(ip, 443, 4.0);
                results.Add(new DiscordPingResult
                {
                    Label = label,
                    Ip = ip,
                    Ok = ok,
                    LatencyMs = ok ? lat : null,
                    Error = err
                });
                progress?.Invoke((double)(i + 1) / DiscordPingHosts.Length);
            }
            return results;
        }

        public static List<BlockType> ClassifyBlocks(DiagReport r)
        {
            var blocks = new HashSet<BlockType>();
            var dcOk = r.DcResults.Where(x => x.Ok).ToList();
            var pingOk = r.PingResults.Where(x => x.Ok).ToList();

            if (r.DnsResult?.Spoofed == true) 
                blocks.Add(BlockType.DnsSpoof);

            if (r.DpiResult?.DpiDetected == true) 
                blocks.Add(BlockType.SniBlock);

            if (dcOk.Count == 0 && pingOk.Count > 0 && !blocks.Contains(BlockType.SniBlock))
                blocks.Add(BlockType.IpBlock);

            if (dcOk.Count > 0 && pingOk.Count > 0)
            {
                var withLat = dcOk.Where(x => x.LatencyMs.HasValue).ToList();
                if (withLat.Count > 0)
                {
                    var avgDc = withLat.Average(x => x.LatencyMs!.Value);
                    var pingWithLat = pingOk.Where(x => x.LatencyMs.HasValue).ToList();
                    if (pingWithLat.Count > 0)
                    {
                        var avgPing = pingWithLat.Average(x => x.LatencyMs!.Value);
                        if ((avgDc > 150 && avgDc > avgPing * 3) || avgDc > 250)
                            blocks.Add(BlockType.Throttling);
                    }
                }
            }

            if (r.MediaResult?.Throttled == true) 
                blocks.Add(BlockType.MediaThrottle);

            return [.. blocks];
        }

        public static List<string> MakeRecommendations(DiagReport r)
        {
            var recs = new List<string>();
            var blocks = r.BlockTypes;
            var app = r.AppStatus ?? new AppStatus();
            var bypass = app.ZapretRunning;

            string BypassList()
            {
                var t = new List<string>();
                if (app.ZapretRunning) t.Add($"Zapret ({app.ZapretProcName})");
                return string.Join(", ", t);
            }

            if (blocks.Contains(BlockType.MediaThrottle))
            {
                var spd = r.MediaResult?.SpeedKbps > 0 ? $"{r.MediaResult.SpeedKbps:F0} kbps" : "не измерена";
                recs.Add($"⚠️  ТСПУ-ЗАМЕДЛЕНИЕ МЕДИА (РКН)\n    Измеренная скорость: {spd} (норма > 200 kbps)\n    Файлы и медиа грузятся медленно. Используйте Zapret или VPN.");
            }

            if (blocks.Contains(BlockType.SniBlock))
                recs.Add("🛡  SNI-БЛОКИРОВКА (DPI обрывает TLS по SNI)\n    Проверьте, что Zapret запущен с нужным профилем.");

            if (blocks.Contains(BlockType.IpBlock))
                recs.Add("📡  IP-БЛОКИРОВКА (все DC недоступны)\n    Требуется VPN или WARP.");

            if (blocks.Contains(BlockType.DnsSpoof))
                recs.Add("🔵  DNS-СПУФИНГ\n    Включите зашифрованный DNS в Windows (1.1.1.1).");

            if (r.DiscordPing != null && r.DiscordPing.Count > 0 && r.DiscordPing.All(p => !p.Ok))
                recs.Add("🔴  DISCORD API: Недоступен!\n    Текстовые чаты и авторизация не будут работать без обхода.");

            if (r.UdpResult?.Blocked == true)
                recs.Add(bypass
                    ? $"🎮  Discord UDP: заблокирован, НО {BypassList()} активен, Discord работает! ✓"
                    : "🎮  Discord UDP заблокирован. Звонки не работают без Zapret.");
            else
                recs.Add("🎮  Discord UDP: доступен ✓, голос работает без дополнительных настроек.");

            return recs;
        }

        public static async Task<DiagReport> RunFullDiagnosticsAsync(Action<double, string>? progress = null)
        {
            void Step(double r, string lbl) => progress?.Invoke(r, lbl);
            var report = new DiagReport();

            Step(0.00, "Определение процессов...");
            report.AppStatus = await Task.Run(CheckAppStatus);

            Step(0.10, "Проверка DC Telegram...");
            report.DcResults = await CheckTelegramDcsAsync(p => Step(0.10 + p * 0.20, "DC Telegram..."));

            Step(0.30, "DNS и DPI...");
            report.DnsResult = await CheckDnsAsync(progress: p => Step(0.30 + p * 0.10, "DNS..."));
            report.DpiResult = await CheckDpiAsync(p => Step(0.40 + p * 0.15, "DPI/SNI..."));

            Step(0.55, "Скорость медиа (ТСПУ)...");
            report.MediaResult = await CheckMediaThrottleAsync(p => Step(0.55 + p * 0.15, "Медиа..."));

            Step(0.70, "UDP порты Discord...");
            report.UdpResult = await CheckDiscordUdpAsync(p => Step(0.70 + p * 0.10, "UDP..."));

            Step(0.80, "Пинг серверов Discord...");
            report.DiscordPing = await CheckDiscordPingAsync(p => Step(0.80 + p * 0.10, "Пинг Discord..."));

            Step(0.90, "Анализ данных...");
            report.BlockTypes = ClassifyBlocks(report);
            report.Recommendations = MakeRecommendations(report);

            Step(1.0, "Готово ✓");
            return report;
        }
    }

using System;

using System.Threading.Tasks;

namespace ZapretGUI.Core
{
    public class ModManager
    {
        private readonly string _modsBasePath;
        private readonly string _strategiesPath;
        private readonly string _listsPath;
        private readonly string _zapretFilesPath;

        public ModManager()
        {
            _modsBasePath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.ModsDirectory);
            _strategiesPath = Path.Combine(_modsBasePath, "strategies");
            _listsPath = Path.Combine(_modsBasePath, "lists");
            _zapretFilesPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.CoreFilesDirectory);
        }

        public void InitializeFolders()
        {
            if (!Directory.Exists(_modsBasePath)) 
                Directory.CreateDirectory(_modsBasePath);

            if (!Directory.Exists(_strategiesPath)) 
                Directory.CreateDirectory(_strategiesPath);

            if (!Directory.Exists(_listsPath)) 
                Directory.CreateDirectory(_listsPath);
        }

        public bool IsModIdTaken(string id)
        {
            if (string.IsNullOrWhiteSpace(id)) return true;
            InitializeFolders();

            var normId = SanitizeModId(id);
            if (Directory.Exists(Path.Combine(_strategiesPath, normId)) ||
                Directory.Exists(Path.Combine(_listsPath, normId)))
            {
                return true;
            }

            foreach (var dir in Directory.GetDirectories(_strategiesPath).Concat(Directory.GetDirectories(_listsPath)))
            {
                var jsonPath = Path.Combine(dir, "mod.json");
                if (File.Exists(jsonPath))
                {
                    try
                    {
                        var meta = JsonSerializer.Deserialize<ModMetaData>(File.ReadAllText(jsonPath), new JsonSerializerOptions { PropertyNameCaseInsensitive = true });
                        if (meta != null && string.Equals(meta.UniqueId, normId, StringComparison.OrdinalIgnoreCase))
                            return true;
                    }
                    catch { }
                }
            }

            return false;
        }

        public static string SanitizeModId(string raw)
        {
            if (string.IsNullOrWhiteSpace(raw)) 
                return "";

            var clean = raw.Trim().ToLowerInvariant().Replace(" ", "_");
            clean = Regex.Replace(clean, @"[^a-z0-9_\-]", "");
            return clean.Trim('_');
        }

        public string GenerateUniqueModId(string baseName = "mod")
        {
            var prefix = SanitizeModId(baseName);
            if (string.IsNullOrEmpty(prefix)) 
                prefix = "mod";

            if (prefix.Length > 18) 
                prefix = prefix.Substring(0, 18).TrimEnd('_');

            while (true)
            {
                var shortHash = Guid.NewGuid().ToString("N").Substring(0, 4);
                var candidate = $"{prefix}_{shortHash}";
                if (!IsModIdTaken(candidate))
                    return candidate;
            }
        }

        public async Task<List<UIModItem>> GetAvailableModsAsync(ModType type)
        {
            var mods = new List<UIModItem>();
            var targetFolder = type == ModType.BatStrategy ? _strategiesPath : _listsPath;

            if (!Directory.Exists(targetFolder))
                return mods;

            var activeList = type == ModType.BatStrategy
                ? SettingsManager.Current.ActiveBatMods
                : SettingsManager.Current.ActiveListMods;

            foreach (var dir in Directory.GetDirectories(targetFolder))
            {
                var modId = Path.GetFileName(dir);
                var jsonPath = Path.Combine(dir, "mod.json");

                if (File.Exists(jsonPath))
                {
                    try
                    {
                        var json = await File.ReadAllTextAsync(jsonPath);
                        var meta = JsonSerializer.Deserialize<ModMetaData>(json, new JsonSerializerOptions { PropertyNameCaseInsensitive = true });

                        if (meta != null)
                        {
                            if (string.IsNullOrWhiteSpace(meta.UniqueId))
                                meta.UniqueId = modId;

                            mods.Add(new UIModItem
                            {
                                Id = modId,
                                Type = type,
                                Meta = meta,
                                IsActive = activeList.Contains(modId)
                            });
                        }
                    }
                    catch (Exception ex)
                    {
                        Debug.WriteLine($"Ошибка чтения meta-данных мода {modId}: {ex.Message}");
                    }
                }
            }
            return mods;
        }

        public async Task ApplyListModsAsync()
        {
            var listsDir = Path.Combine(_zapretFilesPath, "lists");
            var targetFile = Path.Combine(listsDir, "list-general.txt");
            var backupBaseFile = Path.Combine(listsDir, "list-general.base.txt");

            if (!File.Exists(targetFile)) return;

            var allModDomains = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            var activeModDomains = new HashSet<string>(StringComparer.OrdinalIgnoreCase);

            if (Directory.Exists(_listsPath))
            {
                foreach (var dir in Directory.GetDirectories(_listsPath))
                {
                    var modId = Path.GetFileName(dir);
                    var listPath = Path.Combine(dir, "list.txt");

                    if (File.Exists(listPath))
                    {
                        var lines = await File.ReadAllLinesAsync(listPath);
                        var domains = lines.Where(l => !string.IsNullOrWhiteSpace(l) && !l.TrimStart().StartsWith("#"))
                                           .Select(l => l.Trim());

                        foreach (var d in domains)
                        {
                            allModDomains.Add(d);
                            if (SettingsManager.Current.ActiveListMods.Contains(modId))
                                activeModDomains.Add(d);
                        }
                    }
                }
            }

            if (!File.Exists(backupBaseFile))
            {
                var currentRaw = await File.ReadAllLinesAsync(targetFile);
                var baseLinesOnly = currentRaw.Where(l => !string.IsNullOrWhiteSpace(l))
                                              .Select(l => l.Trim())
                                              .Where(l => !allModDomains.Contains(l))
                                              .Distinct(StringComparer.OrdinalIgnoreCase)
                                              .ToList();
                await File.WriteAllLinesAsync(backupBaseFile, baseLinesOnly);
            }

            var baseLines = (await File.ReadAllLinesAsync(backupBaseFile))
                            .Where(l => !string.IsNullOrWhiteSpace(l))
                            .Select(l => l.Trim())
                            .ToList();

            baseLines.AddRange(activeModDomains);

            await File.WriteAllLinesAsync(targetFile, baseLines.Distinct(StringComparer.OrdinalIgnoreCase));
        }

        public async Task SyncActiveBatModsAsync()
        {
            if (!Directory.Exists(_zapretFilesPath)) return;

            await Task.Run(() =>
            {
                foreach (var file in Directory.GetFiles(_zapretFilesPath, "mod_*.bat"))
                    try { File.Delete(file); } catch { }

                foreach (var modId in SettingsManager.Current.ActiveBatMods)
                {
                    var sourceBat = Path.Combine(_strategiesPath, modId, "strategy.bat");
                    if (File.Exists(sourceBat))
                    {
                        var destBat = Path.Combine(_zapretFilesPath, $"mod_{modId}.bat");
                        try { File.Copy(sourceBat, destBat, true); } catch { }
                    }
                }
            });
        }

        public (bool isValid, string errorMessage) ValidateBatMod(string modName, string batContent)
        {
            try
            {
                var lines = (batContent ?? "").Split(new[] { "\r\n", "\n" }, StringSplitOptions.None);
                ZapretManager.ParseAndValidateBatLines(lines, $"моде «{modName}»", _zapretFilesPath, checkReferencedFiles: true);
                return (true, "");
            }
            catch (Exception ex)
            {
                return (false, ex.Message);
            }
        }

            {
                using var client = new TcpClient();
                var stopwatch = new Stopwatch();

                stopwatch.Start();
                var connectTask = client.ConnectAsync(host, port);

                if (await Task.WhenAny(connectTask, Task.Delay(2000)) == connectTask)

                return -1;

            }
        }
    }

using System;
using System.Diagnostics;
using System.Net.NetworkInformation;
using System.Timers;

namespace ZapretGUI.Core
{
    public class NetworkMonitor
    {
        private readonly System.Timers.Timer _networkTimer;
        private readonly Stopwatch _elapsedStopwatch = new Stopwatch();
        private long _lastBytesReceived = 0;
        private long _lastBytesSent = 0;
        private bool _wasNetworkAvailable = true;

        public event Action<double, double>? StatsUpdated;
        public event Action<bool>? StatusChanged;

        public NetworkMonitor()
        {
            _networkTimer = new System.Timers.Timer();
            UpdateIntervalFromSettings();
            _networkTimer.Elapsed += Timer_Tick;

            SettingsManager.SettingsSaved += UpdateIntervalFromSettings;
        }

        public void UpdateIntervalFromSettings()
        {
            int seconds = Math.Max(1, SettingsManager.Current.StatsUpdateInterval);
            _networkTimer.Interval = seconds * 1000;
        }

        public void Start()
        {
            UpdateIntervalFromSettings();
            _elapsedStopwatch.Restart();
            _networkTimer.Start();
        }

        public void Stop()
        {
            _networkTimer.Stop();
            _elapsedStopwatch.Stop();
        }

        private void Timer_Tick(object? sender, ElapsedEventArgs e)
        {
            var isAvailable = NetworkInterface.GetIsNetworkAvailable();

            if (isAvailable != _wasNetworkAvailable)
            {
                _wasNetworkAvailable = isAvailable;
                StatusChanged?.Invoke(isAvailable);
            }

            var currentReceived = 0L;
            var currentSent = 0L;

            var interfaces = NetworkInterface.GetAllNetworkInterfaces();
            foreach (var netInterface in interfaces)
            {
                if (netInterface.OperationalStatus == OperationalStatus.Up &&
                    netInterface.NetworkInterfaceType != NetworkInterfaceType.Loopback)
                {
                    var stats = netInterface.GetIPv4Statistics();
                    currentReceived += stats.BytesReceived;
                    currentSent += stats.BytesSent;
                }
            }

            var elapsedSeconds = Math.Max(0.1, _elapsedStopwatch.Elapsed.TotalSeconds);
            _elapsedStopwatch.Restart();

            if (_lastBytesReceived != 0 && _lastBytesSent != 0)
            {
                var diffReceived = Math.Max(0L, currentReceived - _lastBytesReceived);
                var diffSent = Math.Max(0L, currentSent - _lastBytesSent);

                var mbpsReceived = (diffReceived * 8.0) / (1000000.0 * elapsedSeconds);
                var mbpsSent = (diffSent * 8.0) / (1000000.0 * elapsedSeconds);

                StatsUpdated?.Invoke(mbpsReceived, mbpsSent);
            }

            _lastBytesReceived = currentReceived;
            _lastBytesSent = currentSent;
        }
    }
}

        public static bool ValidatePortRange(string? input, out string cleanRange)
        {
            cleanRange = "";
            if (string.IsNullOrWhiteSpace(input)) return false;

            string trimmed = input.Replace(" ", "");
            var parts = trimmed.Split(',', StringSplitOptions.RemoveEmptyEntries);
            if (parts.Length == 0) return false;

            foreach (var part in parts)
            {
                if (!Regex.IsMatch(part, @"^[1-9][0-9]*(?:-[1-9][0-9]*)?$"))
                    return false;

                var bounds = part.Split('-');
                if (!int.TryParse(bounds[0], out int start) || start < 1 || start > 65535)
                    return false;

                if (bounds.Length == 2)
                {
                    if (!int.TryParse(bounds[1], out int end) || end < 1 || end > 65535 || start > end)
                        return false;
                }
            }

            cleanRange = trimmed;
            return true;
        }

        public static (bool isApplied, int lineCount) GetHostsStatus()
        {
            try
            {
                if (!File.Exists(SystemHostsPath)) return (false, 0);
                var text = File.ReadAllText(SystemHostsPath);

                int startIdx = text.IndexOf(HostsMarkerStart, StringComparison.OrdinalIgnoreCase);
                int endIdx = text.IndexOf(HostsMarkerEnd, StringComparison.OrdinalIgnoreCase);

                if (startIdx >= 0 && endIdx > startIdx)
                {
                    string block = text.Substring(startIdx + HostsMarkerStart.Length, endIdx - (startIdx + HostsMarkerStart.Length));
                    int count = block.Split(new[] { '\r', '\n' }, StringSplitOptions.RemoveEmptyEntries)
                                     .Count(l => !string.IsNullOrWhiteSpace(l) && !l.TrimStart().StartsWith("#"));
                    return (true, count);
                }
                return (false, 0);
            }
            catch
            {
                return (false, 0);
            }
        }

        public static async Task<int> UpdateSystemHostsAsync(int sourceIndex)
        {
            var urls = new List<string>();
            if (sourceIndex == 0) 
                urls.Add(AppConstants.FlowsealHostsUrl);
            else if (sourceIndex == 1 || sourceIndex == 3) 
                urls.Add(AppConstants.V3nillaHostsUrl);
            else
            {
                urls.Add(AppConstants.FlowsealHostsUrl);
                urls.Add(AppConstants.V3nillaHostsUrl);
            }

            var includeAdBlockZeroIps = (sourceIndex == 3);

            var collectedLines = new List<string>();
            var seenEntries = new HashSet<string>(StringComparer.OrdinalIgnoreCase);

            foreach (var url in urls)
            {
                var raw = await _http.GetStringAsync($"{url}?t={Guid.NewGuid():N}");
                foreach (var line in raw.Split(new[] { "\r\n", "\n" }, StringSplitOptions.None))
                {
                    var trimmed = line.Trim();
                    if (string.IsNullOrWhiteSpace(trimmed)) continue;

                    if (!includeAdBlockZeroIps && trimmed.Contains("Блокировка рекламы, трекеров", StringComparison.OrdinalIgnoreCase))
                        break;

                    if (trimmed.StartsWith("#"))
                    {
                        collectedLines.Add(trimmed);
                        continue;
                    }

                    if (trimmed.Contains("localhost", StringComparison.OrdinalIgnoreCase))
                        continue;

                    if (!includeAdBlockZeroIps && trimmed.StartsWith("0.0.0.0"))
                        continue;

                    var parts = trimmed.Split(new[] { ' ', '\t' }, StringSplitOptions.RemoveEmptyEntries);
                    if (parts.Length >= 2)
                    {
                        var normalizedEntry = $"{parts[0]} {parts[1]}";
                        if (seenEntries.Add(normalizedEntry))
                            collectedLines.Add(normalizedEntry);
                    }
                }
            }

            var activeDomainsCount = collectedLines.Count(l => !l.StartsWith("#"));
            if (activeDomainsCount == 0)
                throw new Exception("Не удалось получить ни одной записи из выбранного источника.");

            var backupPath = SystemHostsPath + ".zapret.bak";
            if (File.Exists(SystemHostsPath) && !File.Exists(backupPath))
                try { File.Copy(SystemHostsPath, backupPath, overwrite: false); } catch { }

            var existingContent = File.Exists(SystemHostsPath) ? await File.ReadAllTextAsync(SystemHostsPath) : "";
            var cleanBase = RemoveZapretBlock(existingContent).TrimEnd();

            var sb = new StringBuilder();
            if (!string.IsNullOrEmpty(cleanBase))
            {
                sb.AppendLine(cleanBase);
                sb.AppendLine();
            }

            sb.AppendLine(HostsMarkerStart);
            sb.AppendLine($"# Обновлено: {DateTime.Now:dd.MM.yyyy HH:mm} (Активных записей: {activeDomainsCount})");
            foreach (var l in collectedLines)
                sb.AppendLine(l);
            sb.AppendLine(HostsMarkerEnd);

            await File.WriteAllTextAsync(SystemHostsPath, sb.ToString(), new UTF8Encoding(false));
            FlushDnsCache();

            return activeDomainsCount;
        }

        public static void ClearZapretHosts()
        {
            if (!File.Exists(SystemHostsPath)) return;

            var existingContent = File.ReadAllText(SystemHostsPath);
            var cleanBase = RemoveZapretBlock(existingContent).TrimEnd() + "\r\n";

            File.WriteAllText(SystemHostsPath, cleanBase, new UTF8Encoding(false));
            FlushDnsCache();
        }

        private static string RemoveZapretBlock(string content)
        {
            var startIdx = content.IndexOf(HostsMarkerStart, StringComparison.OrdinalIgnoreCase);
            var endIdx = content.IndexOf(HostsMarkerEnd, StringComparison.OrdinalIgnoreCase);

            if (startIdx >= 0 && endIdx > startIdx)
            {
                var removeLength = (endIdx + HostsMarkerEnd.Length) - startIdx;
                return content.Remove(startIdx, removeLength);
            }
            return content;
        }

        private static void FlushDnsCache()
        {
            try
            {
                Process.Start(new ProcessStartInfo
                {
                    FileName = "ipconfig.exe",
                    Arguments = "/flushdns",
                    CreateNoWindow = true,
                    UseShellExecute = false
                })?.WaitForExit(3000);
            }
            catch { }
        }
    }

        public static bool IsAutoStartEnabled()
        {
            try
            {
                using var key = Registry.CurrentUser.OpenSubKey(RegistryKeyPath, false);
                return key?.GetValue(AppConstants.AppRegistryName) != null;
            }
            catch (Exception ex)
            {
                Debug.WriteLine($"Ошибка чтения реестра: {ex.Message}");
                return false;
            }
        }

        public static void SetAutoStart(bool enable)
        {
            try
            {
                using var key = Registry.CurrentUser.OpenSubKey(RegistryKeyPath, true);
                if (key == null) return;

                if (enable)
                {
                    var exePath = Environment.ProcessPath;
                    if (string.IsNullOrEmpty(exePath))
                        exePath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "ZapretGUI.exe");

                    key.SetValue(AppConstants.AppRegistryName, $"\"{exePath}\"");
                }
                else
                {
                    key.DeleteValue(AppConstants.AppRegistryName, false);
                }
            }
            catch (Exception ex)
            {
                Debug.WriteLine($"Ошибка записи в реестр: {ex.Message}");
            }
        }
    }
}

using System;
using System.Diagnostics;
using System.IO;

namespace ZapretGUI.Core
{
    public class TgProxyManager : IBypassService
    {
        private readonly string _proxyPath;
        private Process? _process;

        public event Action<string>? LogMessage;

        public TgProxyManager()
        {
            _proxyPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.CoreFilesDirectory, $"{AppConstants.TgProxyProcessName}.exe");
        }

        public void Start(string profileName = "")
        {
            if (IsRunning)
                return;

            if (!File.Exists(_proxyPath))
                throw new FileNotFoundException($"Файл прокси не найден: {_proxyPath}");

            var startInfo = new ProcessStartInfo
            {
                FileName = _proxyPath,
                UseShellExecute = false,
                CreateNoWindow = true,
                RedirectStandardOutput = true,
                RedirectStandardError = true
            };

            _process = new Process { StartInfo = startInfo };

            _process.OutputDataReceived += (s, e) => { if (!string.IsNullOrEmpty(e.Data)) LogMessage?.Invoke($"[TgWsProxy] {e.Data}"); };
            _process.ErrorDataReceived += (s, e) => { if (!string.IsNullOrEmpty(e.Data)) LogMessage?.Invoke($"[TgWsProxy ERR] {e.Data}"); };

            _process.Start();
            _process.BeginOutputReadLine();
            _process.BeginErrorReadLine();
        }

        public void Stop()
        {
            try
            {
                if (_process != null && !_process.HasExited)
                {
                    _process.Kill();
                    _process.Dispose();
                    _process = null;
                }
            }
            catch (Exception ex)
            {
                Debug.WriteLine($"Ошибка остановки процесса TgProxy: {ex.Message}");
            }
            ProcessHelper.KillProcessesByName(AppConstants.TgProxyProcessName);
        }

        public bool IsRunning => Process.GetProcessesByName(AppConstants.TgProxyProcessName).Length > 0;
    }
}

                DestroyIcon(_hIcon);

            }

        }

        private static string CleanVersion(string? version)
        {
            if (string.IsNullOrWhiteSpace(version)) return "0.0.0";
            return version.Trim().TrimStart('v', 'V');
        }

        private static bool IsNewerVersion(string? remoteVersion, string? localVersion)
        {
            string cleanRemote = CleanVersion(remoteVersion);
            string cleanLocal = CleanVersion(localVersion);

            if (string.Equals(cleanRemote, cleanLocal, StringComparison.OrdinalIgnoreCase))
                return false;

            if (TryParseNormalizedVersion(cleanRemote, out Version? vRemote) &&
                TryParseNormalizedVersion(cleanLocal, out Version? vLocal))
            {
                return vRemote > vLocal;
            }

            return cleanLocal == "0.0.0" || string.IsNullOrEmpty(cleanLocal);
        }

        private static bool TryParseNormalizedVersion(string input, out Version? version)
        {
            version = null;
            var match = Regex.Match(input, @"^(\d+)(?:\.(\d+))?(?:\.(\d+))?(?:\.(\d+))?");
            if (!match.Success) return false;

            int major = int.Parse(match.Groups[1].Value);
            int minor = match.Groups[2].Success ? int.Parse(match.Groups[2].Value) : 0;
            int build = match.Groups[3].Success ? int.Parse(match.Groups[3].Value) : 0;
            int rev = match.Groups[4].Success ? int.Parse(match.Groups[4].Value) : 0;

            version = new Version(major, minor, build, rev);
            return true;
        }

        public static async Task<AppUpdateResult> CheckForAppUpdateAsync()
        {
            try
            {
                var url = $"https://api.github.com/repos/{AppConstants.RepoOwner}/{AppConstants.RepoName}/releases/latest";
                var response = await _httpClient.GetStringAsync(url);

                using var doc = JsonDocument.Parse(response);
                var latestVersion = doc.RootElement.GetProperty("tag_name").GetString();

                if (latestVersion != null && IsNewerVersion(latestVersion, CurrentVersion))
                {
                    var downloadUrl = "";
                    foreach (var asset in doc.RootElement.GetProperty("assets").EnumerateArray())
                    {
                        var name = asset.GetProperty("name").GetString();
                        if (!string.IsNullOrEmpty(name) && name.EndsWith(".zip", StringComparison.OrdinalIgnoreCase))
                        {
                            downloadUrl = asset.GetProperty("browser_download_url").GetString() ?? "";
                            break;
                        }
                    }

                    return new AppUpdateResult
                    {
                        UpdateAvailable = true,
                        Version = latestVersion,
                        DownloadUrl = downloadUrl,
                        ReleaseUrl = doc.RootElement.GetProperty("html_url").GetString() ?? ""
                    };
                }
            }
            catch (Exception ex)
            {
                Debug.WriteLine($"Ошибка при проверке обновлений GUI: {ex.Message}");
            }

            return new AppUpdateResult { UpdateAvailable = false };
        }

        public static async Task<CoreUpdateResult> CheckForCoreUpdateAsync(string repoUrl, string currentVersion, string coreName, bool isZapret)
        {
            try
            {
                var response = await _httpClient.GetStringAsync(repoUrl);
                using var doc = JsonDocument.Parse(response);

                var latestVersion = doc.RootElement.GetProperty("tag_name").GetString();
                bool filesExist = IsCoreFilePresentOnDisk(isZapret);

                if (latestVersion != null && (!filesExist || IsNewerVersion(latestVersion, currentVersion)))
                {
                    var downloadUrl = "";
                    var targetFileName = "";
                    var isZip = false;

                    foreach (var asset in doc.RootElement.GetProperty("assets").EnumerateArray())
                    {
                        var name = asset.GetProperty("name").GetString();
                        if (string.IsNullOrEmpty(name)) continue;

                        if (isZapret && name.EndsWith(".zip", StringComparison.OrdinalIgnoreCase))
                        {
                            if (string.IsNullOrEmpty(downloadUrl) || name.Contains("winws", StringComparison.OrdinalIgnoreCase))
                            {
                                downloadUrl = asset.GetProperty("browser_download_url").GetString() ?? "";
                                isZip = true;
                            }
                        }
                        else if (!isZapret)
                        {
                            if (name.EndsWith(".exe", StringComparison.OrdinalIgnoreCase) && name.Contains("windows", StringComparison.OrdinalIgnoreCase))
                            {
                                downloadUrl = asset.GetProperty("browser_download_url").GetString() ?? "";
                                targetFileName = $"{AppConstants.TgProxyProcessName}.exe";
                                isZip = false;
                                break;
                            }
                            else if (name.EndsWith(".zip", StringComparison.OrdinalIgnoreCase))
                            {
                                downloadUrl = asset.GetProperty("browser_download_url").GetString() ?? "";
                                isZip = true;
                                break;
                            }
                        }
                    }

                    if (!string.IsNullOrEmpty(downloadUrl))
                    {
                        return new CoreUpdateResult
                        {
                            UpdateAvailable = true,
                            Version = latestVersion,
                            CurrentVersion = currentVersion,
                            CoreName = coreName,
                            DownloadUrl = downloadUrl,
                            TargetFileName = targetFileName,
                            IsZip = isZip
                        };
                    }
                }
            }
            catch (Exception ex)
            {
                Debug.WriteLine($"Ошибка при проверке обновлений {coreName}: {ex.Message}");
            }

            return new CoreUpdateResult { UpdateAvailable = false };
        }

        public static async Task InstallCoreAsync(CoreUpdateResult updateInfo, Action? stopServicesCallback, IProgress<string>? progress = null)
        {
            if (string.IsNullOrEmpty(updateInfo.DownloadUrl))
                throw new Exception("Подходящий файл релиза не найден.");

            stopServicesCallback?.Invoke();

            ProcessHelper.KillProcessesByName(AppConstants.ZapretProcessName);
            ProcessHelper.KillProcessesByName(AppConstants.TgProxyProcessName);

            await Task.Delay(500);

            var extractPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.CoreFilesDirectory);
            Directory.CreateDirectory(extractPath);

            if (updateInfo.IsZip)
            {
                var tempZip = Path.Combine(Path.GetTempPath(), $"{updateInfo.CoreName}_update.zip");

                progress?.Report($"Загрузка {updateInfo.CoreName}...");
                await DownloadFileWithProgressAsync(updateInfo.DownloadUrl, tempZip, progress);
                progress?.Report("Распаковка архива...");

                using (var archive = ZipFile.OpenRead(tempZip))
                {
                    string? rootDirToStrip = null;
                    var hasFilesAtRoot = false;

                    foreach (var entry in archive.Entries)
                    {
                        if (string.IsNullOrEmpty(entry.FullName))
                            continue;

                        var slashIndex = entry.FullName.IndexOf('/');

                        if (slashIndex == -1)
                        {
                            hasFilesAtRoot = true;
                            break;
                        }
                        var currentRoot = entry.FullName.Substring(0, slashIndex + 1);

                        if (rootDirToStrip == null)
                            rootDirToStrip = currentRoot;
                        else if (rootDirToStrip != currentRoot)
                        {
                            rootDirToStrip = null;
                            break;
                        }
                    }

                    if (hasFilesAtRoot)
                        rootDirToStrip = null;

                    foreach (var entry in archive.Entries)
                    {
                        var entryName = entry.FullName;
                        if (rootDirToStrip != null && entryName.StartsWith(rootDirToStrip))
                            entryName = entryName.Substring(rootDirToStrip.Length);

                        if (string.IsNullOrEmpty(entryName) || string.IsNullOrEmpty(entry.Name))
                            continue;

                        var destinationPath = Path.GetFullPath(Path.Combine(extractPath, entryName));
                        Directory.CreateDirectory(Path.GetDirectoryName(destinationPath)!);

                        entry.ExtractToFile(destinationPath, overwrite: true);
                        File.SetLastWriteTime(destinationPath, DateTime.Now);
                    }
                }
                File.Delete(tempZip);
            }
            else
            {
                var destFilePath = Path.Combine(extractPath, updateInfo.TargetFileName);
                progress?.Report($"Загрузка {updateInfo.CoreName}...");
                await DownloadFileWithProgressAsync(updateInfo.DownloadUrl, destFilePath, progress);
            }

            progress?.Report("✅ Обновление успешно завершено.");
        }

        public static async Task ApplyAppUpdateAsync(string downloadUrl, Action shutdownAppCallback)
        {
            var tempDir = Path.Combine(Path.GetTempPath(), "ZapretGUI_Update");
            var tempZip = Path.Combine(tempDir, "update.zip");
            var extractDir = Path.Combine(tempDir, "Extracted");

            if (Directory.Exists(tempDir))
                Directory.Delete(tempDir, true);

            Directory.CreateDirectory(tempDir);
            Directory.CreateDirectory(extractDir);

            await DownloadFileWithProgressAsync(downloadUrl, tempZip, null);
            ZipFile.ExtractToDirectory(tempZip, extractDir, overwriteFiles: true);

            var extractedExe = Directory.GetFiles(extractDir, "*.exe").FirstOrDefault();
            var sourceExePath = extractedExe ?? Path.Combine(extractDir, "ZapretGUI.exe");

            var currentAppDir = AppDomain.CurrentDomain.BaseDirectory;
            var currentExePath = Environment.ProcessPath;
            if (string.IsNullOrEmpty(currentExePath))
            {
                currentExePath = Path.Combine(currentAppDir, "ZapretGUI.exe");
            }

            var batPath = Path.Combine(tempDir, "updater.bat");
            var batContent = $@"@echo off
chcp 65001 > nul
echo Обновление панели управления... Пожалуйста, подождите.
timeout /t 3 /nobreak > nul
copy /Y ""{sourceExePath}"" ""{currentExePath}""
if not exist ""{Path.Combine(currentAppDir, AppConstants.CoreFilesDirectory)}"" (
    xcopy /Y /E /I ""{Path.Combine(extractDir, AppConstants.CoreFilesDirectory)}"" ""{Path.Combine(currentAppDir, AppConstants.CoreFilesDirectory)}""
)
start """" ""{currentExePath}""
rmdir /S /Q ""{tempDir}""
del ""%~f0""
";
            File.WriteAllText(batPath, batContent, System.Text.Encoding.UTF8);

            Process.Start(new ProcessStartInfo { FileName = batPath, UseShellExecute = true, WindowStyle = ProcessWindowStyle.Hidden });

            shutdownAppCallback?.Invoke();
        }

        private static async Task DownloadFileWithProgressAsync(string downloadUrl, string destinationFilePath, IProgress<string>? progress)
        {
            using var response = await _httpClient.GetAsync(downloadUrl, HttpCompletionOption.ResponseHeadersRead);
            response.EnsureSuccessStatusCode();

            var totalBytes = response.Content.Headers.ContentLength ?? -1L;
            var canReportProgress = totalBytes != -1;

            using var contentStream = await response.Content.ReadAsStreamAsync();
            using var fileStream = new FileStream(destinationFilePath, FileMode.Create, FileAccess.Write, FileShare.None, 8192, true);

            var buffer = new byte[8192];
            var isMoreToRead = true;
            var totalRead = 0L;

            do
            {
                var read = await contentStream.ReadAsync(buffer, 0, buffer.Length);
                if (read == 0)
                    isMoreToRead = false;
                else
                {
                    await fileStream.WriteAsync(buffer, 0, read);
                    totalRead += read;

                    if (canReportProgress)
                    {
                        var percentage = Math.Round((double)totalRead / totalBytes * 100, 1);
                        progress?.Report($"Скачивание: {percentage}%");
                    }
                    else
                        progress?.Report($"Скачано: {totalRead / 1024 / 1024} МБ");
                }
            }
            while (isMoreToRead);
        }

using System;

namespace ZapretGUI.Core
{
    public class WebBridge
    {
        private readonly WebView2 _webView;
        private readonly Window _mainWindow;
        private readonly ModManager _modManager = new ModManager();

        private FileSystemWatcher? _modsWatcher;
        private CancellationTokenSource? _watcherDebounceCts;
        private bool _isInternalFileOp = false;

        private static readonly JsonSerializerOptions _jsonOptions = new JsonSerializerOptions
        {
            PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
            PropertyNameCaseInsensitive = true
        };

        public WebBridge(WebView2 webView, Window mainWindow)
        {
            _webView = webView;
            _mainWindow = mainWindow;

            _modManager.InitializeFolders();
            SetupLiveModsWatcher();

            BypassController.Current.OnLog += msg => SendEvent("log", new
            {
                message = msg,
                time = DateTime.Now.ToString("HH:mm:ss"),
                isProgress = msg.Contains("Скачивание:") || msg.Contains("Скачано:")
            });

            BypassController.Current.NetMonitor.StatsUpdated += (rx, tx) =>
            {
                SendEvent("netStats", new
                {
                    download = Math.Round(rx, 1),
                    upload = Math.Round(tx, 1),
                    zapretRunning = BypassController.Current.Zapret.IsRunning,
                    proxyRunning = BypassController.Current.TgProxy.IsRunning
                });
            };

            BypassController.Current.NetMonitor.StatusChanged += isOnline =>
            {
                SendEvent("netStatus", new { isOnline });
            };

            BypassController.Current.Scanner.ConfigTested += (file, ok, total) =>
            {
                SendEvent("configTested", new { fileName = file, ok, total });
            };

            BypassController.Current.Scanner.ScanCompleted += topConfigs =>
            {
                if (topConfigs.Count > 0)
                {
                    SettingsManager.Current.SelectedProfileName = topConfigs[0];
                    SettingsManager.Save();
                }
                SendEvent("scanCompleted", new
                {
                    topConfigs,
                    bestConfig = topConfigs.FirstOrDefault() ?? ""
                });
            };

            BypassController.Current.Scanner.ScanFailed += ex =>
            {
                SendEvent("scanFailed", new { error = ex.Message });
            };
        }

        private void SetupLiveModsWatcher()
        {
            var modsPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.ModsDirectory);
            if (!Directory.Exists(modsPath)) Directory.CreateDirectory(modsPath);

            _modsWatcher = new FileSystemWatcher(modsPath)
            {
                IncludeSubdirectories = true,
                NotifyFilter = NotifyFilters.FileName | NotifyFilters.DirectoryName | NotifyFilters.LastWrite,
                EnableRaisingEvents = true
            };

            FileSystemEventHandler handler = (s, e) =>
            {
                if (_isInternalFileOp) return;
                _watcherDebounceCts?.Cancel();
                _watcherDebounceCts = new CancellationTokenSource();
                var token = _watcherDebounceCts.Token;

                Task.Run(async () =>
                {
                    try
                    {
                        await Task.Delay(350, token);
                        if (token.IsCancellationRequested) return;
                        await _modManager.ApplyListModsAsync();
                        await _modManager.SyncActiveBatModsAsync();
                        await BroadcastModsAndProfilesAsync();
                    }
                    catch { }
                });
            };

            _modsWatcher.Changed += handler;
            _modsWatcher.Created += handler;
            _modsWatcher.Deleted += handler;
            _modsWatcher.Renamed += (s, e) => handler(s, e);
        }

        public void SendEvent(string eventName, object payload)
        {
            try
            {
                _mainWindow.Dispatcher.InvokeAsync(() =>
                {
                    if (_webView.CoreWebView2 != null)
                    {
                        var json = JsonSerializer.Serialize(new { type = eventName, data = payload }, _jsonOptions);
                        _webView.CoreWebView2.PostWebMessageAsJson(json);
                    }
                });
            }
            catch (Exception ex)
            {
                Debug.WriteLine($"Ошибка отправки в WebView2: {ex.Message}");
            }
        }

        public async void HandleMessage(string rawJson)
        {
            try
            {
                using var doc = JsonDocument.Parse(rawJson);
                var root = doc.RootElement;

                if (root.ValueKind == JsonValueKind.String)
                {
                    using var innerDoc = JsonDocument.Parse(root.GetString() ?? "{}");
                    await ProcessActionAsync(innerDoc.RootElement.Clone());
                }
                else
                {
                    await ProcessActionAsync(root.Clone());
                }
            }
            catch (Exception ex)
            {
                SendEvent("log", new { message = $"❌ ОШИБКА: {ex.Message}", time = DateTime.Now.ToString("HH:mm:ss") });
            }
        }

        private async Task ProcessActionAsync(JsonElement root)
        {
            if (!root.TryGetProperty("action", out var actionProp)) return;
            var action = actionProp.GetString() ?? "";

            switch (action)
            {
                case "init":
                    await SendInitialStateAsync();
                    break;

                case "windowDrag":
                    _mainWindow.Dispatcher.Invoke(() => { try { _mainWindow.DragMove(); } catch { } });
                    break;

                case "windowMinimize":
                    _mainWindow.Dispatcher.Invoke(() => _mainWindow.WindowState = WindowState.Minimized);
                    break;

                case "windowClose":
                    _mainWindow.Dispatcher.Invoke(() => _mainWindow.Close());
                    break;

                case "playClick":
                    AudioHelper.PlayClick();
                    break;

                case "playSuccess":
                    AudioHelper.PlaySuccess();
                    break;

                case "openUrl":
                    var url = root.GetProperty("url").GetString() ?? "";
                    if (!string.IsNullOrWhiteSpace(url))
                        Process.Start(new ProcessStartInfo(url) { UseShellExecute = true });
                    break;

                case "toggleBypass":
                    await HandleToggleBypassAsync(root);
                    break;

                case "restartBypass":
                    await BypassController.Current.RestartServicesAsync(SettingsManager.Current.SelectedProfileName);
                    SendEvent("bypassState", new
                    {
                        isRunning = BypassController.Current.IsRunning,
                        zapretRunning = BypassController.Current.Zapret.IsRunning,
                        proxyRunning = BypassController.Current.TgProxy.IsRunning
                    });
                    break;

                case "updateModuleToggles":
                    if (root.TryGetProperty("useZapret", out var uz)) SettingsManager.Current.ZapretEnabled = uz.GetBoolean();
                    if (root.TryGetProperty("useProxy", out var up)) SettingsManager.Current.TgProxyEnabled = up.GetBoolean();
                    SettingsManager.Save();
                    break;

                case "selectProfile":
                    var prof = root.GetProperty("profile").GetString() ?? "general.bat";
                    SettingsManager.Current.SelectedProfileName = prof;
                    SettingsManager.Save();
                    SendEvent("profileSelected", new { profile = prof });
                    break;

                case "ping":
                    var pingMs = await BypassController.Current.CheckPingAsync();
                    SendEvent("pingResult", new { ping = pingMs });
                    break;

                case "startScan":
                    SettingsManager.Current.FirstTimeScanPrompted = true;
                    SettingsManager.Save();

                    if (BypassController.Current.Scanner.IsScanning)
                    {
                        BypassController.Current.Scanner.CancelScan();
                        SendEvent("scanState", new { isScanning = false });
                    }
                    else
                    {
                        SendEvent("scanState", new { isScanning = true });
                        await BypassController.Current.Scanner.StartScanAsync();
                        SendEvent("scanState", new { isScanning = false });
                    }
                    break;

                case "markFirstScanPrompted":
                    SettingsManager.Current.FirstTimeScanPrompted = true;
                    SettingsManager.Save();
                    break;

                case "saveGameFilter":
                    await HandleSaveGameFilterAsync(root);
                    break;

                case "exportLogs":
                    HandleExportLogs(root);
                    break;

                // ==================== КОМАНДЫ МОДИФИКАЦИЙ (MODSVIEW) ====================
                case "getMods":
                    await BroadcastModsAndProfilesAsync();
                    break;

                case "generateModId":
                    var baseName = root.TryGetProperty("baseName", out var bn) ? bn.GetString() ?? "mod" : "mod";
                    var genId = _modManager.GenerateUniqueModId(baseName);
                    SendEvent("modIdGenerated", new { id = genId, isTaken = false });
                    break;

                case "checkModId":
                    var candidate = root.GetProperty("id").GetString() ?? "";
                    var cleanId = ModManager.SanitizeModId(candidate);
                    var isTaken = string.IsNullOrEmpty(cleanId) || _modManager.IsModIdTaken(cleanId);
                    SendEvent("modIdChecked", new { cleanId, isTaken, isEmpty = string.IsNullOrEmpty(cleanId) });
                    break;

                case "createMod":
                    await HandleCreateModAsync(root);
                    break;

                case "toggleMod":
                    await HandleToggleModAsync(root);
                    break;

                case "disableAllMods":
                    await HandleDisableAllModsAsync(root);
                    break;

                case "deleteMod":
                    await HandleDeleteModAsync(root);
                    break;

                case "getModContent":
                    await HandleGetModContentAsync(root);
                    break;

                case "validateBatMod":
                    var modTitle = root.GetProperty("name").GetString() ?? "Скрипт";
                    var batCode = root.GetProperty("content").GetString() ?? "";
                    var (isValid, errMsg) = _modManager.ValidateBatMod(modTitle, batCode);
                    SendEvent("batValidationResult", new { isValid, errorMessage = errMsg });
                    break;

                case "saveModContent":
                    await HandleSaveModContentAsync(root);
                    break;

                case "importMod":
                    await HandleImportModAsync();
                    break;

                case "exportMod":
                    await HandleExportModAsync(root);
                    break;

                // ==================== ДИАГНОСТИКА (DIAGNOSTICSVIEW) ====================
                case "runDiagnostics":
                    await HandleRunDiagnosticsAsync();
                    break;

                // ==================== НАСТРОЙКИ (SETTINGSVIEW) ====================
                case "saveSettings":
                    HandleSaveSettings(root.GetProperty("settings"));
                    break;

                case "setIpSetMode":
                    HandleSetIpSetMode(root);
                    break;

                case "updateIpSet":
                    await HandleUpdateIpSetAsync();
                    break;

                case "updateHosts":
                    await HandleUpdateHostsAsync(root);
                    break;

                case "clearHosts":
                    HandleClearHosts();
                    break;

                case "checkUpdates":
                    await HandleCheckUpdatesManualAsync();
                    break;

                case "resetSettings":
                    await HandleResetSettingsAsync();
                    break;

                case "openFolder":
                    var folderType = root.GetProperty("folder").GetString();
                    var targetPath = folderType switch
                    {
                        "mods" => Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.ModsDirectory),
                        "zapret" => Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.CoreFilesDirectory),
                        _ => AppDomain.CurrentDomain.BaseDirectory
                    };
                    if (!Directory.Exists(targetPath)) Directory.CreateDirectory(targetPath);
                    Process.Start("explorer.exe", targetPath);
                    break;
            }
        }

        private async Task HandleCreateModAsync(JsonElement root)
        {
            var isBat = root.GetProperty("isBat").GetBoolean();
            var rawId = root.GetProperty("id").GetString() ?? "";
            var name = root.GetProperty("name").GetString() ?? "";
            var author = root.GetProperty("author").GetString() ?? "Аноним";
            var desc = root.GetProperty("description").GetString() ?? "";

            var id = ModManager.SanitizeModId(rawId);
            if (string.IsNullOrEmpty(id) || _modManager.IsModIdTaken(id))
            {
                SendEvent("alert", new { title = "Конфликт ID", message = "Мод с таким ID уже существует! Нажмите 🎲 для генерации уникального ID.", isError = true });
                return;
            }

            var folderName = isBat ? "strategies" : "lists";
            var modFolderPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.ModsDirectory, folderName, id);

            try
            {
                _isInternalFileOp = true;
                Directory.CreateDirectory(modFolderPath);

                var meta = new ModMetaData
                {
                    UniqueId = id,
                    Name = string.IsNullOrWhiteSpace(name) ? id : name.Trim(),
                    Author = string.IsNullOrWhiteSpace(author) ? "Аноним" : author.Trim(),
                    Version = "1.0",
                    Description = desc.Trim(),
                    IsBatStrategy = isBat
                };

                var jsonPath = Path.Combine(modFolderPath, "mod.json");
                await File.WriteAllTextAsync(jsonPath, JsonSerializer.Serialize(meta, new JsonSerializerOptions { WriteIndented = true }));

                if (isBat)
                {
                    var defaultBat = ":: Шаблон стратегии обхода\r\n" +
                                     "\"%BIN%winws.exe\" --wf-tcp=80,443 --wf-udp=443,50000-50100 ^\r\n" +
                                     "--filter-udp=443 --hostlist=\"%LISTS%list-general.txt\" --dpi-desync=fake --dpi-desync-repeats=6 ^\r\n" +
                                     "--new ^\r\n" +
                                     "--filter-tcp=80,443 --hostlist=\"%LISTS%list-general.txt\" --dpi-desync=fake,split2\r\n";
                    await File.WriteAllTextAsync(Path.Combine(modFolderPath, "strategy.bat"), defaultBat);
                }
                else
                {
                    await File.WriteAllTextAsync(Path.Combine(modFolderPath, "list.txt"), "");
                }

                SendEvent("modCreated", new { id, isBat });
                await BroadcastModsAndProfilesAsync();
            }
            catch (Exception ex)
            {
                SendEvent("alert", new { title = "Ошибка", message = $"Ошибка при создании мода: {ex.Message}", isError = true });
            }
            finally
            {
                _isInternalFileOp = false;
            }
        }

        private async Task HandleToggleModAsync(JsonElement root)
        {
            var id = root.GetProperty("id").GetString() ?? "";
            var isBat = root.GetProperty("isBat").GetBoolean();
            var modName = root.TryGetProperty("name", out var n) ? n.GetString() ?? id : id;

            try
            {
                _isInternalFileOp = true;
                var list = isBat ? SettingsManager.Current.ActiveBatMods : SettingsManager.Current.ActiveListMods;
                bool nowActive;

                if (list.Contains(id))
                {
                    list.Remove(id);
                    nowActive = false;
                }
                else
                {
                    list.Add(id);
                    nowActive = true;
                }

                if (isBat)
                    await _modManager.SyncActiveBatModsAsync();
                else
                    await _modManager.ApplyListModsAsync();

                SettingsManager.Save();
                await BroadcastModsAndProfilesAsync();

                if (!isBat && BypassController.Current.Zapret.IsRunning)
                {
                    var word = nowActive ? "включен" : "отключен";
                    SendEvent("promptRestartZapret", new { reason = $"Список доменов «{modName}» был {word}." });
                }
            }
            finally
            {
                _isInternalFileOp = false;
            }
        }

            }
            finally
            {
                _isInternalFileOp = false;
            }
        }

        private async Task HandleDeleteModAsync(JsonElement root)
        {
            var id = root.GetProperty("id").GetString() ?? "";
            var isBat = root.GetProperty("isBat").GetBoolean();
            var modName = root.TryGetProperty("name", out var n) ? n.GetString() ?? id : id;

            try
            {
                _isInternalFileOp = true;
                var list = isBat ? SettingsManager.Current.ActiveBatMods : SettingsManager.Current.ActiveListMods;
                var wasActiveList = !isBat && list.Contains(id);

                var folderName = isBat ? "strategies" : "lists";
                var modFolderPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.ModsDirectory, folderName, id);

                if (Directory.Exists(modFolderPath))
                    Directory.Delete(modFolderPath, true);

                if (list.Remove(id))
                {
                    if (isBat) await _modManager.SyncActiveBatModsAsync();
                    else await _modManager.ApplyListModsAsync();
                    SettingsManager.Save();
                }

                await BroadcastModsAndProfilesAsync();

                if (wasActiveList && BypassController.Current.Zapret.IsRunning)
                    SendEvent("promptRestartZapret", new { reason = $"Активный список доменов «{modName}» был удален." });
            }
            catch (Exception ex)
            {
                SendEvent("alert", new { title = "Ошибка удаления", message = $"Не удалось удалить мод: {ex.Message}", isError = true });
            }
            finally
            {
                _isInternalFileOp = false;
            }
        }

        private async Task HandleGetModContentAsync(JsonElement root)
        {
            var id = root.GetProperty("id").GetString() ?? "";
            var isBat = root.GetProperty("isBat").GetBoolean();

            var folderName = isBat ? "strategies" : "lists";
            var fileName = isBat ? "strategy.bat" : "list.txt";
            var modFolder = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.ModsDirectory, folderName, id);
            var filePath = Path.Combine(modFolder, fileName);
            var jsonPath = Path.Combine(modFolder, "mod.json");

            if (!File.Exists(filePath))
            {
                SendEvent("alert", new { title = "Ошибка", message = $"Файл {fileName} не найден в папке мода!", isError = true });
                return;
            }

            var content = await File.ReadAllTextAsync(filePath);
            ModMetaData meta = new ModMetaData { UniqueId = id, Name = id, IsBatStrategy = isBat };
            if (File.Exists(jsonPath))
            {
                try
                {
                    var m = JsonSerializer.Deserialize<ModMetaData>(await File.ReadAllTextAsync(jsonPath), _jsonOptions);
                    if (m != null) meta = m;
                }
                catch { }
            }

            SendEvent("modContentLoaded", new
            {
                id,
                isBat,
                fileName,
                meta,
                content
            });
        }

        private async Task HandleSaveModContentAsync(JsonElement root)
        {
            var id = root.GetProperty("id").GetString() ?? "";
            var isBat = root.GetProperty("isBat").GetBoolean();
            var name = root.GetProperty("name").GetString() ?? id;
            var author = root.GetProperty("author").GetString() ?? "Аноним";
            var version = root.GetProperty("version").GetString() ?? "1.0";
            var desc = root.GetProperty("description").GetString() ?? "";
            var content = root.GetProperty("content").GetString() ?? "";
            var bypassWarning = root.TryGetProperty("bypassWarning", out var bw) && bw.GetBoolean();

            var editedModName = string.IsNullOrWhiteSpace(name) ? id : name.Trim();

            if (isBat && !bypassWarning)
            {
                var (isValid, errMsg) = _modManager.ValidateBatMod(editedModName, content);
                if (!isValid)
                {
                    SendEvent("modSaveSyntaxWarning", new { errorMessage = errMsg });
                    return;
                }
            }

            try
            {
                _isInternalFileOp = true;
                var folderName = isBat ? "strategies" : "lists";
                var fileName = isBat ? "strategy.bat" : "list.txt";
                var modFolder = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.ModsDirectory, folderName, id);

                var meta = new ModMetaData
                {
                    UniqueId = id,
                    Name = editedModName,
                    Author = string.IsNullOrWhiteSpace(author) ? "Аноним" : author.Trim(),
                    Version = string.IsNullOrWhiteSpace(version) ? "1.0" : version.Trim(),
                    Description = desc.Trim(),
                    IsBatStrategy = isBat
                };

                await File.WriteAllTextAsync(Path.Combine(modFolder, "mod.json"), JsonSerializer.Serialize(meta, new JsonSerializerOptions { WriteIndented = true }));
                await File.WriteAllTextAsync(Path.Combine(modFolder, fileName), content);

                if (isBat) await _modManager.SyncActiveBatModsAsync();
                else await _modManager.ApplyListModsAsync();

                SettingsManager.Save();
                SendEvent("modContentSaved", new { id });
                await BroadcastModsAndProfilesAsync();

                bool isActiveList = !isBat && SettingsManager.Current.ActiveListMods.Contains(id);
                if (isActiveList && BypassController.Current.Zapret.IsRunning)
                {
                    SendEvent("promptRestartZapret", new { reason = $"Содержимое активного списка «{editedModName}» было обновлено." });
                }
            }
            catch (Exception ex)
            {
                SendEvent("alert", new { title = "Ошибка сохранения", message = ex.Message, isError = true });
            }
            finally
            {
                _isInternalFileOp = false;
            }
        }

        private async Task HandleImportModAsync()
        {
            string? selectedFile = null;
            _mainWindow.Dispatcher.Invoke(() =>
            {
                var openFileDialog = new Microsoft.Win32.OpenFileDialog
                {
                    Filter = "Архивы модов (*.zip;*.netfix-mod)|*.zip;*.netfix-mod|Все файлы (*.*)|*.*",
                    Title = "Выберите архив с модом"
                };
                if (openFileDialog.ShowDialog() == true)
                    selectedFile = openFileDialog.FileName;
            });

            if (string.IsNullOrEmpty(selectedFile)) return;

            var tempDir = Path.Combine(Path.GetTempPath(), "ZapretGUI_Mod_" + Guid.NewGuid().ToString("N"));
            try
            {
                _isInternalFileOp = true;
                Directory.CreateDirectory(tempDir);
                await Task.Run(() => ZipFile.ExtractToDirectory(selectedFile, tempDir));

                var jsonFiles = Directory.GetFiles(tempDir, "mod.json", SearchOption.AllDirectories);
                if (jsonFiles.Length == 0)
                {
                    SendEvent("alert", new { title = "Ошибка импорта", message = "В архиве не найден файл mod.json. Убедитесь, что это корректный мод.", isError = true });
                    return;
                }

                var jsonPath = jsonFiles[0];
                var modRootPath = Path.GetDirectoryName(jsonPath)!;

                var meta = JsonSerializer.Deserialize<ModMetaData>(await File.ReadAllTextAsync(jsonPath), _jsonOptions);
                if (meta == null)
                {
                    SendEvent("alert", new { title = "Ошибка импорта", message = "Файл mod.json поврежден.", isError = true });
                    return;
                }

                var baseCandidate = !string.IsNullOrWhiteSpace(meta.UniqueId)
                    ? ModManager.SanitizeModId(meta.UniqueId)
                    : ModManager.SanitizeModId(meta.Name);

                if (string.IsNullOrEmpty(baseCandidate))
                    baseCandidate = ModManager.SanitizeModId(Path.GetFileNameWithoutExtension(selectedFile));

                var finalUniqueId = _modManager.IsModIdTaken(baseCandidate)
                    ? _modManager.GenerateUniqueModId(baseCandidate)
                    : baseCandidate;

                meta.UniqueId = finalUniqueId;
                await File.WriteAllTextAsync(jsonPath, JsonSerializer.Serialize(meta, new JsonSerializerOptions { WriteIndented = true }));

                var folderType = meta.IsBatStrategy ? "strategies" : "lists";
                var destPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.ModsDirectory, folderType, finalUniqueId);
                Directory.CreateDirectory(destPath);

                await Task.Run(() =>
                {
                    foreach (var file in Directory.GetFiles(modRootPath, "*.*", SearchOption.AllDirectories))
                    {
                        var relativePath = file.Substring(modRootPath.Length + 1);
                        var targetPath = Path.Combine(destPath, relativePath);
                        Directory.CreateDirectory(Path.GetDirectoryName(targetPath)!);
                        File.Copy(file, targetPath, true);
                    }
                });

                await BroadcastModsAndProfilesAsync();
                SendEvent("alert", new { title = "Импорт завершен", message = $"Мод «{meta.Name}» (ID: {finalUniqueId}) успешно импортирован!", isError = false });
            }
            catch (Exception ex)
            {
                SendEvent("alert", new { title = "Ошибка импорта", message = ex.Message, isError = true });
            }
            finally
            {
                _isInternalFileOp = false;
                if (Directory.Exists(tempDir))
                    try { Directory.Delete(tempDir, true); } catch { }
            }
        }

        private async Task HandleExportModAsync(JsonElement root)
        {
            var id = root.GetProperty("id").GetString() ?? "";
            var isBat = root.GetProperty("isBat").GetBoolean();
            var modName = root.TryGetProperty("name", out var n) ? n.GetString() ?? id : id;

            string? savePath = null;
            _mainWindow.Dispatcher.Invoke(() =>
            {
                var saveFileDialog = new Microsoft.Win32.SaveFileDialog
                {
                    Filter = "Архивы модов (*.zip)|*.zip|Мод NetFix (*.netfix-mod)|*.netfix-mod|Все файлы (*.*)|*.*",
                    Title = "Экспорт мода",
                    FileName = $"{id}_export.zip"
                };
                if (saveFileDialog.ShowDialog() == true)
                    savePath = saveFileDialog.FileName;
            });

            if (string.IsNullOrEmpty(savePath)) return;

            try
            {
                var folderName = isBat ? "strategies" : "lists";
                var modFolderPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.ModsDirectory, folderName, id);

                if (!Directory.Exists(modFolderPath))
                {
                    SendEvent("alert", new { title = "Ошибка", message = "Папка мода не найдена на диске!", isError = true });
                    return;
                }

                if (File.Exists(savePath)) File.Delete(savePath);
                await Task.Run(() => ZipFile.CreateFromDirectory(modFolderPath, savePath));

                SendEvent("alert", new { title = "Экспорт завершен", message = $"Мод «{modName}» (ID: {id}) успешно экспортирован и готов к отправке!", isError = false });
            }
            catch (Exception ex)
            {
                SendEvent("alert", new { title = "Ошибка экспорта", message = ex.Message, isError = true });
            }
        }

        private List<string> GetBatProfiles()
        {
            var folderPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.CoreFilesDirectory);
            if (!Directory.Exists(folderPath)) return new List<string> { "general.bat" };

            return Directory.GetFiles(folderPath, "*.bat")
                .Select(Path.GetFileName)
                .Where(name => !string.IsNullOrEmpty(name) &&
                              (name.StartsWith("general", StringComparison.OrdinalIgnoreCase) ||
                               name.StartsWith("mod_", StringComparison.OrdinalIgnoreCase)))
                .Cast<string>()
                .OrderBy(n => n)
                .ToList();
        }

                return;
            }

            if (!ServiceToolsManager.ValidatePortRange(udpRaw, out string validUdp))
            {
                SendEvent("alert", new { title = "Ошибка портов UDP", message = "Указан некорректный диапазон портов UDP!\nПример правильного формата: 1024-65535", isError = true });
                return;
            }

            SettingsManager.Current.GameFilterMode = mode;
            SettingsManager.Current.GameFilterTcpRange = validTcp;
            SettingsManager.Current.GameFilterUdpRange = validUdp;

            ServiceToolsManager.SyncGameFilterFile();
            SettingsManager.Save();

            var statusText = ServiceToolsManager.GetGameFilterDisplayStatus();
            SendEvent("gameFilterSaved", new
            {
                mode,
                tcp = validTcp,
                udp = validUdp,
                statusText
            });

            SendEvent("log", new { message = $"🎮 Игровой фильтр обновлен: {statusText}", time = DateTime.Now.ToString("HH:mm:ss") });

            if (restartNow && BypassController.Current.Zapret.IsRunning)
            {
                await BypassController.Current.RestartServicesAsync(SettingsManager.Current.SelectedProfileName);
                SendEvent("bypassState", new
                {
                    isRunning = BypassController.Current.IsRunning,
                    zapretRunning = BypassController.Current.Zapret.IsRunning,
                    proxyRunning = BypassController.Current.TgProxy.IsRunning

            }
        }

        }

using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.IO;
using System.Linq;
using System.Text.Json;
using System.Text.RegularExpressions;

namespace ZapretGUI.Core
{
    public class ZapretManager : IBypassService
    {
        private readonly string _basePath;
        private Process? _process;

        public event Action<string>? LogMessage;


        public ZapretManager()
        {
            _basePath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.CoreFilesDirectory);
        }

        private string GetSourceDisplayName(string profileName)
        {
            if (profileName.StartsWith("mod_", StringComparison.OrdinalIgnoreCase))
            {
                var modId = Path.GetFileNameWithoutExtension(profileName).Substring(4);
                var modJsonPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.ModsDirectory, "strategies", modId, "mod.json");
                if (File.Exists(modJsonPath))
                {
                    try
                    {
                        var meta = JsonSerializer.Deserialize<ModMetaData>(File.ReadAllText(modJsonPath), new JsonSerializerOptions { PropertyNameCaseInsensitive = true });
                        if (meta != null && !string.IsNullOrWhiteSpace(meta.Name))
                            return $"моде «{meta.Name}» ({profileName})";
                    }
                    catch { }
                }
                return $"моде «{modId}» ({profileName})";
            }
            return $"профиле «{profileName}»";
        }

        public static string ParseAndValidateBatLines(string[] lines, string sourceName, string zapretBasePath, bool checkReferencedFiles = true)
        {
            if (lines == null || lines.Length == 0 || lines.All(string.IsNullOrWhiteSpace))
                throw new Exception($"Ошибка в {sourceName}: Файл стратегии пуст.");

            var arguments = "";
            var isReadingArgs = false;
            var winwsFound = false;
            var winwsStartLine = 0;

            for (var i = 0; i < lines.Length; i++)
            {
                var lineNum = i + 1;
                var rawLine = lines[i];
                var trimmed = rawLine.Trim();

                if (string.IsNullOrWhiteSpace(trimmed) ||
                    trimmed.StartsWith("::") ||
                    trimmed.StartsWith("rem ", StringComparison.OrdinalIgnoreCase) ||
                    trimmed.Equals("rem", StringComparison.OrdinalIgnoreCase))
                {
                    continue;
                }

                if (!isReadingArgs)
                {
                    if (trimmed.Contains(AppConstants.ZapretProcessName, StringComparison.OrdinalIgnoreCase))
                    {
                        isReadingArgs = true;
                        winwsFound = true;
                        winwsStartLine = lineNum;

                        var match = Regex.Match(trimmed, $@"{AppConstants.ZapretProcessName}(?:\.exe)?[""']?\s*(.*)", RegexOptions.IgnoreCase);
                        trimmed = match.Success ? match.Groups[1].Value.Trim() : "";
                    }
                    else
                    {
                        var isStandardBatCmd = Regex.IsMatch(trimmed, @"^(@?echo|chcp|cd|pushd|popd|set|title|cls|color|mode|goto|call|if|for|pause|timeout|:)\b", RegexOptions.IgnoreCase);
                        if (!isStandardBatCmd)
                            throw new Exception($"Синтаксическая ошибка в {sourceName} [Строка {lineNum}]:\nНеизвестная команда «{trimmed}».\nОжидался вызов {AppConstants.ZapretProcessName}.exe или стандартная инструкция Batch.");
                        continue;
                    }
                }

                if (isReadingArgs)
                {
                    var hasContinuation = trimmed.EndsWith("^");
                    var cleanLine = hasContinuation ? trimmed.Substring(0, trimmed.Length - 1).TrimEnd() : trimmed;
                    var quoteCount = cleanLine.Count(c => c == '\"');

                    if (quoteCount % 2 != 0)
                        throw new Exception($"Ошибка синтаксиса в {sourceName} [Строка {lineNum}]:\nНе закрыта кавычка (\") в параметрах:\n«{rawLine.Trim()}»");

                    if (!string.IsNullOrWhiteSpace(cleanLine))
                    {
                        var tokens = Regex.Matches(cleanLine, @"[\""].+?[\""]|[^ ]+")
                                          .Select(m => m.Value.Trim())
                                          .ToList();

                        foreach (var token in tokens)
                        {
                            if (!token.StartsWith("--") && !token.StartsWith("-") && !token.Equals("--new", StringComparison.OrdinalIgnoreCase))
                                throw new Exception($"Ошибка параметра в {sourceName} [Строка {lineNum}]:\nАргумент «{token}» не подходит для Zapret (winws).\nВсе параметры должны начинаться с «--» (например, --filter-tcp=80 или --dpi-desync=fake).\nСтрока: «{rawLine.Trim()}»");

                            if (token.StartsWith("---") || token == "--" || token == "-")
                                throw new Exception($"Ошибка синтаксиса в {sourceName} [Строка {lineNum}]:\nНекорректный формат флага «{token}».\nСтрока: «{rawLine.Trim()}»");

                            if (checkReferencedFiles && Directory.Exists(zapretBasePath))
                            {
                                var fileMatch = Regex.Match(token, @"^--(hostlist|hostlist-exclude|ipset|ipset-exclude|dpi-desync-fake-tls|dpi-desync-fake-quic|dpi-desync-fake-unknown-udp|dpi-desync-split-seqovl-pattern)=(.+)$", RegexOptions.IgnoreCase);
                                if (fileMatch.Success)
                                {
                                    var paramName = fileMatch.Groups[1].Value;
                                    var rawFilePath = fileMatch.Groups[2].Value.Trim('\"', '\'');

                                    if (!rawFilePath.StartsWith("0x", StringComparison.OrdinalIgnoreCase) && !rawFilePath.StartsWith("!"))
                                    {
                                        var resolvedPath = rawFilePath
                                            .Replace("%BIN%", @"bin\")
                                            .Replace("%LISTS%", @"lists\")
                                            .Replace("%~dp0", "");

                                        var fullCheckPath = Path.IsPathRooted(resolvedPath)
                                            ? resolvedPath
                                            : Path.Combine(zapretBasePath, resolvedPath);

                                        if (!File.Exists(fullCheckPath))
                                            throw new Exception($"Ошибка в {sourceName} [Строка {lineNum}]:\nВ параметре «--{paramName}» указан несуществующий файл «{rawFilePath}».\nСтрока: «{rawLine.Trim()}»");
                                    }
                                }
                            }
                        }

                        arguments += " " + cleanLine;
                    }

                    if (!hasContinuation)
                        break;
                }
            }

            if (!winwsFound)
                throw new Exception($"Ошибка структуры в {sourceName}:\nВ батнике не найден вызов ядра «{AppConstants.ZapretProcessName}.exe». Этот скрипт не подходит для Zapret.");

            var (gfAll, gfTcp, gfUdp) = ServiceToolsManager.GetGameFilterSubstitutions();
            ServiceToolsManager.SyncGameFilterFile();

            arguments = arguments.Replace("%BIN%", @"bin\")
                                 .Replace("%LISTS%", @"lists\")
                                 .Replace("%~dp0", "")
                                 .Replace("%GameFilterTCP%", gfTcp)
                                 .Replace("%GameFilterUDP%", gfUdp)
                                 .Replace("%GameFilter%", gfAll)
                                 .Trim();

            if (string.IsNullOrEmpty(arguments))
                throw new Exception($"Ошибка в {sourceName} [Строка {winwsStartLine}]:\nКоманда «{AppConstants.ZapretProcessName}» вызвана без параметров фильтрации.");

            var unexpandedVar = Regex.Match(arguments, @"%[a-zA-Z0-9_]+%");
            if (unexpandedVar.Success)
                throw new Exception($"Ошибка в {sourceName}:\nОбнаружена неизвестная переменная «{unexpandedVar.Value}», которая не поддерживается в прямом запуске Zapret.");

            return arguments;
        }

        public void Start(string profileName)
        {
            Stop();

            var batFilePath = Path.Combine(_basePath, profileName);
            var sourceName = GetSourceDisplayName(profileName);

            if (!File.Exists(batFilePath))
                throw new FileNotFoundException($"Конфигурационный файл не найден в {sourceName} по пути: {batFilePath}");

            var lines = File.ReadAllLines(batFilePath);
            var arguments = ParseAndValidateBatLines(lines, sourceName, _basePath, checkReferencedFiles: true);

            var exePath = Path.Combine(_basePath, $"{AppConstants.ZapretProcessName}.exe");
            var binExePath = Path.Combine(_basePath, "bin", $"{AppConstants.ZapretProcessName}.exe");

            if (!File.Exists(exePath) && File.Exists(binExePath))
                exePath = binExePath;

            if (!File.Exists(exePath))
                throw new FileNotFoundException($"Не найден исполняемый файл ядра по пути: {exePath}. Запустите проверку обновлений в настройках.");

            var startInfo = new ProcessStartInfo
            {
                FileName = exePath,
                Arguments = arguments,
                UseShellExecute = false,
                CreateNoWindow = true,
                RedirectStandardOutput = true,
                RedirectStandardError = true,
                WorkingDirectory = _basePath
            };

            _process = new Process { StartInfo = startInfo };

            var errorOutputs = new List<string>();

            _process.OutputDataReceived += (s, e) =>
            {
                if (string.IsNullOrEmpty(e.Data)) 
                    return;

                var line = e.Data;
                var noise = new[]
                {
                    "Loading hostlist", "loading plain text list", "Loaded",
                    "user defined desync profile", "github version", "windivert initialized"
                };

                if (noise.Any(n => line.Contains(n, StringComparison.OrdinalIgnoreCase)))
                    return;

                LogMessage?.Invoke($"[Zapret] {line}");
            };

            _process.ErrorDataReceived += (s, e) =>
            {
                if (!string.IsNullOrEmpty(e.Data))
                {
                    errorOutputs.Add(e.Data);
                    LogMessage?.Invoke($"[Zapret ERR] {e.Data}");
                }
            };

            _process.Start();
            _process.BeginOutputReadLine();
            _process.BeginErrorReadLine();

            if (_process.WaitForExit(350))
            {
                var errDetails = errorOutputs.Count > 0
                    ? string.Join(" | ", errorOutputs)
                    : $"Код завершения: {_process.ExitCode}";

                throw new Exception($"Ядро Zapret отвергло конфигурацию в {sourceName}!\nПричина от движка winws: {errDetails}");
            }
        }

        public void Stop()
        {
            try
            {
                if (_process != null && !_process.HasExited)
                {
                    _process.Kill();
                    _process.Dispose();
                    _process = null;
                }
            }
            catch (Exception ex)
            {
                Debug.WriteLine($"Ошибка остановки процесса Zapret: {ex.Message}");
            }

            ProcessHelper.KillProcessesByName(AppConstants.ZapretProcessName);
        }

        public bool IsRunning 
            => Process.GetProcessesByName(AppConstants.ZapretProcessName).Length > 0;
    }

using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.IO;
using System.Runtime.InteropServices;
using System.Text.RegularExpressions;
using System.Threading;
using System.Threading.Tasks;
using System.Timers;

namespace ZapretGUI.Core
{
    public class ZapretScanner
    {
        [DllImport("user32.dll")]
        private static extern bool ShowWindow(IntPtr hWnd, int nCmdShow);
        private const int SW_HIDE = 0;

        private Process? _scanProcess;
        private CancellationTokenSource? _scanCts;
        private readonly List<string> _topConfigs = new List<string>();

        public event Action<string>? LogMessage;
        public event Action<string, int, int>? ConfigTested; // fileName, okCount, totalCount
        public event Action<List<string>>? ScanCompleted;
        public event Action<Exception>? ScanFailed;

        public bool IsScanning => _scanProcess != null && !_scanProcess.HasExited;

        public async Task StartScanAsync()
        {
            if (IsScanning) return;

            _topConfigs.Clear();
            _scanCts = new CancellationTokenSource();

            using var hideTimer = new System.Timers.Timer(100);
            hideTimer.Elapsed += (s, ev) =>
            {
                foreach (var p in Process.GetProcessesByName(AppConstants.ZapretProcessName))
                {
                    if (p.MainWindowHandle != IntPtr.Zero)
                        ShowWindow(p.MainWindowHandle, SW_HIDE);
                }
            };
            hideTimer.Start();

            try
            {
                var scriptPath = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, AppConstants.CoreFilesDirectory, "utils", "test zapret.ps1");

                if (!File.Exists(scriptPath))
                    throw new FileNotFoundException("Скрипт тестирования не найден по пути: " + scriptPath);

                var startInfo = new ProcessStartInfo
                {
                    FileName = "powershell.exe",
                    Arguments = $"-NoProfile -WindowStyle Hidden -ExecutionPolicy Bypass -File \"{scriptPath}\"",
                    UseShellExecute = false,
                    RedirectStandardOutput = true,
                    RedirectStandardInput = true,
                    RedirectStandardError = true,
                    CreateNoWindow = true,
                    StandardOutputEncoding = System.Text.Encoding.UTF8
                };

                _scanProcess = new Process { StartInfo = startInfo };
                _scanProcess.Start();

                await _scanProcess.StandardInput.WriteLineAsync("1");
                await _scanProcess.StandardInput.WriteLineAsync("1");
                _scanProcess.StandardInput.Close();

                await Task.Run(async () =>
                {
                    while (!_scanProcess.StandardOutput.EndOfStream)
                    {
                        if (_scanCts.Token.IsCancellationRequested) break;

                        var line = await _scanProcess.StandardOutput.ReadLineAsync();
                        if (string.IsNullOrWhiteSpace(line))
                            continue;

                        line = Regex.Replace(line, @"\x1B\[[0-9;]*[a-zA-Z]", "");

                        string? simpleLog = SimplifyLogLine(line);
                        if (simpleLog != null)
                            LogMessage?.Invoke($"[Auto] {simpleLog}");
                    }

                    if (!_scanProcess.HasExited)
                        _scanProcess.WaitForExit();

                }, _scanCts.Token);

                if (_scanCts.Token.IsCancellationRequested)
                    LogMessage?.Invoke("🛑 Сканирование было прервано пользователем.");
                else
                    ScanCompleted?.Invoke(new List<string>(_topConfigs));
            }
            catch (Exception ex)
            {
                ScanFailed?.Invoke(ex);
            }
            finally
            {
                hideTimer.Stop();
            }
        }

        public void CancelScan()
        {
            _scanCts?.Cancel();
            try
            {
                if (_scanProcess != null && !_scanProcess.HasExited)
                {
                    _scanProcess.Kill(entireProcessTree: true);
                    _scanProcess.Dispose();
                    _scanProcess = null;
                }
            }
            catch { }

            ProcessHelper.KillProcessesByName(AppConstants.ZapretProcessName);
        }

        private string? SimplifyLogLine(string line)
        {
            if (string.IsNullOrWhiteSpace(line))
                return null;

            var progressMatch = Regex.Match(line, @"\[(\d+/\d+)\]\s+([a-zA-Z0-9_\-\(\)\s]+\.bat)", RegexOptions.IgnoreCase);
            if (progressMatch.Success)
            {
                var progress = progressMatch.Groups[1].Value;
                var configName = progressMatch.Groups[2].Value.Trim();
                return $"[{progress}] 🔍 Анализ профиля: {configName}...";
            }

            if (line.Contains("=== ANALYTICS ==="))
                return "📊 Сводка результатов тестирования:";

            var statsMatch = Regex.Match(line, @"([a-zA-Z0-9_\-\(\)\s]+\.bat)\s*:\s*(?:HTTP\s*)?OK:\s*(\d+).*?(?:ERR|FAIL):\s*(\d+)", RegexOptions.IgnoreCase);
            if (statsMatch.Success)
            {
                var batName = statsMatch.Groups[1].Value.Trim();
                var okStr = statsMatch.Groups[2].Value;
                var errStr = statsMatch.Groups[3].Value;

                int.TryParse(okStr, out int ok);
                int.TryParse(errStr, out int err);
                ConfigTested?.Invoke(batName, ok, ok + err);

                if (err == 0 && ok > 0)
                {
                    if (!_topConfigs.Contains(batName)) _topConfigs.Add(batName);
                    return $"      ✨ {batName} -> Работает идеально (Успешно: {okStr})";
                }
                else
                    return $"      ⚠️ {batName} -> Есть блокировки (Успешно: {okStr}, Ошибок: {errStr})";
            }

            if (line.Contains("Best config:"))
            {
                var bestMatch = Regex.Match(line, @"Best config:\s*([a-zA-Z0-9_\-\(\)\s]+\.bat)", RegexOptions.IgnoreCase);
                if (bestMatch.Success)
                {
                    var best = bestMatch.Groups[1].Value.Trim();
                    if (_topConfigs.Contains(best)) _topConfigs.Remove(best);
                    _topConfigs.Insert(0, best);
                }
                return null;
            }

            return null;
        }
    }
}

    public partial class TrayMenuWindow : Window
    {
        private bool _isRunning = false;

        public TrayMenuWindow()
        {
            InitializeComponent();
        }

        public void RefreshState()
        {
            SyncStatus();
            CheckPingAsync();
        }

        private void Window_Deactivated(object sender, EventArgs e)
        {
            this.Hide();
        }

        private async void CheckPingAsync()
        {
            PingText.Text = "измерение...";
            PingText.Foreground = new SolidColorBrush(System.Windows.Media.Colors.Gray);

            try
            {
                var pingMs = await BypassController.Current.CheckPingAsync();

                if (pingMs >= 0)
                {
                    PingText.Text = $"{pingMs} мс";

                    if (pingMs < 80)
                        PingText.Foreground = UIHelper.GetSuccessBrush();
                    else if (pingMs < 150)
                        PingText.Foreground = new SolidColorBrush(System.Windows.Media.Colors.Orange);
                    else
                        PingText.Foreground = UIHelper.GetErrorBrush();
                }
                else
                {
                    PingText.Text = "ошибка";
                    PingText.Foreground = UIHelper.GetErrorBrush();
                }
            }
            catch (Exception ex)
            {
                Debug.WriteLine($"Ошибка проверки пинга из трея: {ex.Message}");
                PingText.Text = "нет сети";
                PingText.Foreground = UIHelper.GetErrorBrush();
            }
        }

        private void SyncStatus()
        {
            if (System.Windows.Application.Current.MainWindow is MainWindow mainWindow)
                _isRunning = mainWindow.IsBypassRunning();

            UpdateToggleButtonVisuals();
        }

        private void BtnTogglePower_Click(object sender, RoutedEventArgs e)
        {
            _isRunning = !_isRunning;
            UpdateToggleButtonVisuals();

            if (System.Windows.Application.Current.MainWindow is MainWindow mainWindow)
                mainWindow.ToggleBypass();
        }

        private void UpdateToggleButtonVisuals()
        {
            BtnTogglePower.ApplyTemplate();

            var powerIcon = (System.Windows.Controls.TextBlock)BtnTogglePower.Template.FindName("PowerIcon", BtnTogglePower);
            var powerText = (System.Windows.Controls.TextBlock)BtnTogglePower.Template.FindName("PowerText", BtnTogglePower);

            if (_isRunning)
            {
                BtnTogglePower.Background = UIHelper.GetBrushFromHex("#331A1A");
                BtnTogglePower.BorderBrush = UIHelper.GetBrushFromHex("#5C2E2E");

                if (powerIcon != null)

                    powerIcon.Text = "\xE71A";

                if (powerText != null)

            {

                BtnTogglePower.BorderBrush = UIHelper.GetBrushFromHex("#2E5C37");

                if (powerIcon != null)

                ContentScale.ScaleX = 1;
                ContentScale.ScaleY = 1;

                var fadeIn = new DoubleAnimation(0, 1, TimeSpan.FromMilliseconds(250))
                {
                    EasingFunction = new CubicEase { EasingMode = EasingMode.EaseOut }
                };
                WizardContent.BeginAnimation(OpacityProperty, fadeIn);

                if (isGoingBack)
                {
                    ContentTranslate.X = -30;
                    var slideRight = new DoubleAnimation(-30, 0, TimeSpan.FromMilliseconds(250))
                    {
                        EasingFunction = new CubicEase { EasingMode = EasingMode.EaseOut }
                    };
                    ContentTranslate.BeginAnimation(TranslateTransform.XProperty, slideRight);
                }
                else
                {
                    if (_currentPageIndex == 1)
                    {
                        ContentTranslate.Y = 40;
                        var slideUp = new DoubleAnimation(40, 0, TimeSpan.FromMilliseconds(350))
                        {
                            EasingFunction = new BackEase { Amplitude = 0.3, EasingMode = EasingMode.EaseOut }
                        };
                        ContentTranslate.BeginAnimation(TranslateTransform.YProperty, slideUp);
                    }
                    else if (_currentPageIndex == _pages.Count - 1)
                    {
                        ContentScale.ScaleX = 0.8;
                        ContentScale.ScaleY = 0.8;
                        var zoomIn = new DoubleAnimation(0.8, 1, TimeSpan.FromMilliseconds(350))
                        {
                            EasingFunction = new CircleEase { EasingMode = EasingMode.EaseOut }
                        };
                        ContentScale.BeginAnimation(ScaleTransform.ScaleXProperty, zoomIn);
                        ContentScale.BeginAnimation(ScaleTransform.ScaleYProperty, zoomIn);
                    }
                    else
                    {
                        ContentTranslate.X = 30;
                        var slideLeft = new DoubleAnimation(30, 0, TimeSpan.FromMilliseconds(250))
                        {
                            EasingFunction = new CubicEase { EasingMode = EasingMode.EaseOut }
                        };
                        ContentTranslate.BeginAnimation(TranslateTransform.XProperty, slideLeft);
                    }
                }

                await Task.Delay(350);
                _isAnimating = false;
            }
        }

        private void UpdateProgressDots()
        {
            ProgressDots.Children.Clear();
            for (var i = 0; i < _pages.Count; i++)
            {
                var dot = new Ellipse
                {
                    Width = 8,
                    Height = 8,
                    Margin = new Thickness(4, 0, 4, 0),
                    Fill = i == _currentPageIndex
                        ? new SolidColorBrush((System.Windows.Media.Color)System.Windows.Media.ColorConverter.ConvertFromString("#107C10"))
                        : new SolidColorBrush((System.Windows.Media.Color)System.Windows.Media.ColorConverter.ConvertFromString("#333333"))
                };
                ProgressDots.Children.Add(dot);
            }
        }

        private void BtnCloseWizard_Click(object sender, RoutedEventArgs e)
        {
            this.Close();
        }

        private async void BtnNextStep_Click(object sender, RoutedEventArgs e)
        {
            if (_isAnimating) return;

            if (_currentPageIndex == _pages.Count - 2)
            {
                _currentPageIndex++;
                LoadCurrentPage();

                BtnNextStep.IsEnabled = false;
                BtnCloseWizard.IsEnabled = false;

                var stepServices = _pages[2] as Step4_ServicesConfig;
                var stepSettings = _pages[3] as Step4_AdditionalSettings;
                var stepFinish = _pages[4] as Step5_FinishPage;

                var useZapret = stepServices?.UseZapret ?? true;
                var useTgProxy = stepServices?.UseTgProxy ?? true;

                var autoStart = stepSettings?.IsAutoStart ?? true;
                var focusMode = stepSettings?.IsFocusMode ?? false;
                var colorblind = stepSettings?.IsColorblind ?? false;

                if (stepFinish != null)
                    await stepFinish.RunSetupAsync(useZapret, useTgProxy, autoStart, focusMode, colorblind);

                BtnNextStep.IsEnabled = true;
                BtnNextStep.Content = "Закрыть";
            }
            else if (_currentPageIndex == _pages.Count - 1)
                this.Close();
            else if (_currentPageIndex < _pages.Count - 1)
            {
                _currentPageIndex++;
                LoadCurrentPage();
            }
        }

        private void BtnBack_Click(object sender, RoutedEventArgs e)
        {
            if (_isAnimating) 
                return;

            if (_currentPageIndex > 0)
            {
                _currentPageIndex--;
                LoadCurrentPage(true, true);
            }
        }

    }

using System;
using System.Diagnostics;
using System.Threading.Tasks;
using System.Windows.Controls;
using System.Windows.Media;
using ZapretGUI.Core;

namespace ZapretGUI.Views.WizardPages
{
    public partial class Step5_FinishPage : System.Windows.Controls.UserControl
    {
        private bool _isLastMessageProgress = false;

        public Step5_FinishPage()
        {
            InitializeComponent();
        }

        private async Task AppendLog(string message, int delay = 100)
        {
            bool isProgress = message.Contains("Скачивание:") || message.Contains("Скачано:");

            if (isProgress && _isLastMessageProgress)
            {
                var currentText = ConsoleLog.Text;
                var lastNewLine = currentText.LastIndexOf('\n', Math.Max(0, currentText.Length - 2));

                if (lastNewLine >= 0)
                    ConsoleLog.Text = currentText.Substring(0, lastNewLine + 1);
                else
                    ConsoleLog.Text = "";
            }

            ConsoleLog.Text += $"[{DateTime.Now:HH:mm:ss}] {message}\n";
            LogScroller.ScrollToEnd();

            _isLastMessageProgress = isProgress;

            if (delay > 0)
                await Task.Delay(delay);
        }

        public async Task RunSetupAsync(bool useZapret, bool useTgProxy, bool autoStart, bool focusMode, bool colorblind)
        {
            ConsoleLog.Text = "";

            await AppendLog("Инициализация модуля установки...", 200);

            SettingsManager.Current.ZapretEnabled = useZapret;
            SettingsManager.Current.TgProxyEnabled = useTgProxy;
            SettingsManager.Current.FocusMode = focusMode;
            SettingsManager.Current.ColorblindMode = colorblind;

            await AppendLog("[Config] Сохранение пользовательских настроек UI...", 100);
            SettingsManager.Save();

            if (autoStart)
            {
                await AppendLog("[System] Регистрация приложения в автозагрузке Windows...", 150);
                StartupHelper.SetAutoStart(true);
            }

            await AppendLog("Проверка директорий...", 150);
            var zapretDir = System.IO.Path.Combine(System.AppDomain.CurrentDomain.BaseDirectory, AppConstants.CoreFilesDirectory);

            if (!System.IO.Directory.Exists(zapretDir))
            {
                await AppendLog($"[FS] Создание директории: {zapretDir}", 150);
                System.IO.Directory.CreateDirectory(zapretDir);
            }

            if (useZapret || useTgProxy)
            {
                await AppendLog("=====================================", 0);
                await AppendLog("Запуск загрузки выбранных ядер...", 100);

                var progress = new Progress<string>(status =>
                {
                    _ = AppendLog($"[Download] {status}", 0);
                });

                try
                {
                    await UpdateManager.InstallModulesSilentAsync(useZapret, useTgProxy, progress);
                    await AppendLog("✅ Модули успешно загружены и распакованы.", 200);
                }
                catch (Exception ex)
                {
                    Debug.WriteLine($"Сбой скачивания при начальной настройке: {ex.Message}");
                    await AppendLog($"[ERROR] Ошибка при загрузке: {ex.Message}", 0);
                }

                await AppendLog("=====================================", 0);
            }

            await AppendLog("Сборка завершена.", 400);

            ProgBar.IsIndeterminate = false;
            ProgBar.Value = 100;
            TxtStatus.Text = "Всё готово!";
            TxtStatus.Foreground = UIHelper.GetBrushFromHex("#107C10");

            await AppendLog("✅ Настройка успешно завершена. Можно закрывать окно.", 0);
        }
    }
